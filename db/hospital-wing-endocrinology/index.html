def insulin_correction(current_bg, target_bg, correction_factor):
    """Correction insulin dose estimator (for education; verify with provider)."""
    if current_bg <= target_bg: return "No correction needed"
    units = (current_bg - target_bg) / correction_factor
    return f"Correction: ~{units:.1f} units (1 unit per {correction_factor} mg/dL over target)"
