import hashlib
pairs = [(0.85, 1), (0.8, 1), (0.9, 0), (0.75, 1), (0.88, 1)]
def compute():
    n = len(pairs)
    sq = [(p - y) ** 2 for p, y in pairs]
    brier = round(sum(sq) / n, 4)
    mean_p = round(sum(p for p, _ in pairs) / n, 4)
    hits = sum(1 for _, y in pairs if y == 1)
    out = {"brier": brier, "hits": hits, "mean_p": mean_p, "n": n}
    return out
outputs = compute()
result = "|".join(f"{k}={v}" for k, v in sorted(outputs.items()))
h1 = hashlib.sha256(result.encode()).hexdigest()
outputs2 = compute()
result2 = "|".join(f"{k}={v}" for k, v in sorted(outputs.items()))
h2 = hashlib.sha256(result2.encode()).hexdigest()
for k, v in sorted(outputs.items()):
    print(f"{k}: {v}")
print(f"Result: {result}")
print(f"Run 1 SHA-256: {h1}")
print(f"Run 2 SHA-256: {h2}")
print(f"Deterministic: {'MATCH' if h1 == h2 else 'MISMATCH'}")
