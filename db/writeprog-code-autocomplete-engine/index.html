Working implementation

function trieNew(){return {kids:{},word:false};}
function trieAdd(t,w){var n=t;for(var i=0;i<w.length;i++){var c=w[i];if(!n.kids[c])n.kids[c]=trieNew();n=n.kids[c];}n.word=true;}
function trieSuggest(t,prefix,limit){var n=t;for(var i=0;i<prefix.length;i++){n=n.kids[prefix[i]];if(!n)return [];}var out=[];(function walk(x,acc){if(out.length>=(limit||10))return;if(x.word)out.push(acc);Object.keys(x.kids).sort().forEach(function(k){walk(x.kids[k],acc+k);});})(n,prefix);return out;}
