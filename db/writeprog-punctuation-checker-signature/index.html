Working implementation

function punctCheck(text){var issues=[];var sents=text.match(/[^.!?]+[.!?]+/g)||[];
sents.forEach(function(s,i){var t=s.trim();if(t&&t[0]!==t[0].toUpperCase()&&/[a-zA-Z]/.test(t[0]))issues.push({sent:i+1,rule:'capitalize',msg:'sentence should start with capital'});});
var terms=(text.match(/[.!?]/g)||[]).length;if(sents.length===0&&text.trim())issues.push({sent:1,rule:'terminal',msg:'missing terminal punctuation'});
var dbl=text.match(/([.!?])\1/);if(dbl)issues.push({sent:0,rule:'double',msg:'doubled punctuation "'+dbl[0]+'"'});
return issues;}
