Gauntlet Rank 1 · simple · dungeon adventure · beginner

Gauntlet, released in 1985, is a four-player dungeon crawler in which warriors, a valkyrie, a wizard, and an elf fight through monster-generator rooms, grabbing food and treasure on the way to the exit.

The tribute demo is a dungeon run with monster rooms, food-free treasure, and a clear exit goal on a small grid map: walls block movement, monsters must be attacked from beside them, treasure adds points, and reaching the exit wins the run. Each attack deals damage, and a monster that survives hits back, so health matters. The demo is an original program inspired by the mechanics, not the arcade game itself.

Plan the route before the first step. Clear the monster that guards the corridor, take the treasure on the way rather than doubling back for it, and keep enough health in reserve for the last room. A won run with health left beats a fast run that ends defeated.

Playable tribute note: The demo below is an original tribute inspired by the mechanics of Gauntlet (1985). It is not Gauntlet (1985), and it uses no art, music, or code from that game.

How to play

Type up/down/left/right/attack, quit to stop.

import sys
PARAMS = {'map': ['###########', '#P...#....#', '#....M...E#', '#..T.....#', '#....M...#', '#..........#', '###########'], 'monster_hp': 25}
AUTO = ['down', 'right', 'right', 'right', 'attack', 'right', 'right', 'right', 'right', 'right']

DIRS_D = {"up": (0, -1), "down": (0, 1), "left": (-1, 0), "right": (1, 0)}

def parse_dungeon():
    walls, monsters, treasures, start, exitc = set(), {}, set(), None, None
    for y, row in enumerate(PARAMS["map"]):
        for x, ch in enumerate(row):
            if ch == "#":
                walls.add((x, y))
            elif ch == "M":
                monsters[(x, y)] = PARAMS["monster_hp"]
            elif ch == "T":
                treasures.add((x, y))
            elif ch == "P":
                start = (x, y)
            elif ch == "E":
                exitc = (x, y)
    return walls, monsters, treasures, start, exitc

def dungeon_step(state, cmd):
    walls, monsters, treasures, player, hp, score = state
    if cmd == "attack":
        for (mx, my) in list(monsters):
            if abs(mx - player[0]) + abs(my - player[1]) == 1:
                monsters[(mx, my)] -= 25
                if monsters[(mx, my)] <= 0:
                    del monsters[(mx, my)]
                    score += 30
                else:
                    hp -= 10
                break
        return (walls, monsters, treasures, player, hp, score)
    dx, dy = DIRS_D.get(cmd, (0, 0))
    nxt = (player[0] + dx, player[1] + dy)
    if nxt not in walls and nxt not in monsters:
        player = nxt
    if player in treasures:
        treasures = treasures - {player}
        score += 50
    return (walls, monsters, treasures, player, hp, score)

def dungeon_result(state, exitc):
    walls, monsters, treasures, player, hp, score = state
    if player == exitc:
        return "WIN"
    if hp <= 0:
        return "DEFEATED"
    return "ONGOING"

def auto_play():
    walls, monsters, treasures, start, exitc = parse_dungeon()
    state = (walls, monsters, treasures, start, 100, 0)
    print("Dungeon tribute: auto session")
    for i, cmd in enumerate(AUTO):
        state = dungeon_step(state, cmd)
        walls, monsters, treasures, player, hp, score = state
        print("Step %d: %s pos %s hp %d monsters %d score %d" % (i + 1, cmd, player, hp, len(monsters), score))
        if dungeon_result(state, exitc) != "ONGOING":
            break
    print("FINAL SCORE: %d" % (state[5] + state[4]))
    print("RESULT: %s" % dungeon_result(state, exitc))

def play():
    walls, monsters, treasures, start, exitc = parse_dungeon()
    state = (walls, monsters, treasures, start, 100, 0)
    print("Dungeon tribute: type up/down/left/right/attack, or quit. Reach the exit, grab treasure, defeat monsters.")
    for line in sys.stdin:
        cmd = line.strip().lower()
        if cmd == "quit":
            break
        if cmd not in DIRS_D and cmd != "attack":
            print("Unknown command: %s" % cmd)
            continue
        state = dungeon_step(state, cmd)
        walls, monsters, treasures, player, hp, score = state
        print("pos %s hp %d monsters %d score %d" % (player, hp, len(monsters), score))

            break
    print("FINAL SCORE: %d" % (state[5] + state[4]))
    print("RESULT: %s" % dungeon_result(state, exitc))

if __name__ == "__main__":
    if "--play" in sys.argv:
        play()
    else:
        auto_play()

PARAMS = {'map': ['###########', '#E...#T..P#', '#....M....#', '#.....M..#', '#...T....#', '#..........#', '###########'], 'monster_hp': 50}
AUTO = ['down', 'left', 'left', 'left', 'attack', 'attack', 'left', 'left', 'up', 'left', 'left', 'left']

DIRS_D = {"up": (0, -1), "down": (0, 1), "left": (-1, 0), "right": (1, 0)}

def parse_dungeon():
    walls, monsters, treasures, start, exitc = set(), {}, set(), None, None
    for y, row in enumerate(PARAMS["map"]):
        for x, ch in enumerate(row):
            if ch == "#":
                walls.add((x, y))
            elif ch == "M":
                monsters[(x, y)] = PARAMS["monster_hp"]
            elif ch == "T":
                treasures.add((x, y))
            elif ch == "P":
                start = (x, y)
            elif ch == "E":
                exitc = (x, y)
    return walls, monsters, treasures, start, exitc

def dungeon_step(state, cmd):
    walls, monsters, treasures, player, hp, score = state
    if cmd == "attack":
        for (mx, my) in list(monsters):
            if abs(mx - player[0]) + abs(my - player[1]) == 1:
                monsters[(mx, my)] -= 25
                if monsters[(mx, my)] <= 0:
                    del monsters[(mx, my)]
                    score += 30
                else:
                    hp -= 10
                break
        return (walls, monsters, treasures, player, hp, score)
    dx, dy = DIRS_D.get(cmd, (0, 0))
    nxt = (player[0] + dx, player[1] + dy)
    if nxt not in walls and nxt not in monsters:
        player = nxt
    if player in treasures:
        treasures = treasures - {player}
        score += 50
    return (walls, monsters, treasures, player, hp, score)

def dungeon_result(state, exitc):
    walls, monsters, treasures, player, hp, score = state
    if player == exitc:
        return "WIN"
    if hp <= 0:
        return "DEFEATED"
    return "ONGOING"

def auto_play():
    walls, monsters, treasures, start, exitc = parse_dungeon()

    print("Dungeon tribute: auto session")
    for i, cmd in enumerate(AUTO):
        state = dungeon_step(state, cmd)
        walls, monsters, treasures, player, hp, score = state
        print("Step %d: %s pos %s hp %d monsters %d score %d" % (i + 1, cmd, player, hp, len(monsters), score))
