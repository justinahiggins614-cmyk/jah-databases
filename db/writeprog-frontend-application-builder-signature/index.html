Working implementation

function feComponent(name,renderFn){return {name:name,render:renderFn};}
function feBuild(components,rootName){var out=['// bundle'];
components.forEach(function(c){out.push('function '+c.name+'(props){return ('+c.render.toString()+')(props||{});}');});
out.push('document.body.innerHTML = '+rootName+'();');return out.join('\n');}
function feRenderToString(comp,props){return comp.render(props||{});}
