import json
import random

title = 'Definition of Serenity'
valence = 0.6
arousal = -0.3
plutchik = 3

quadrant = 'low-arousal pleasant'

import hashlib
import json
import random

title = 'The Serenity Field Guide Index'
valence = 0.6
arousal = -0.3
plutchik = 3
seed = 7023
quadrant = 'low-arousal pleasant'

def run_once():
    rng = random.Random(seed)
    jitter = rng.uniform(-0.3, 0.3)
    return round(5 * (valence + arousal) + jitter, 2)

intensity = round(5 * (valence + arousal), 2)
sampled_a = run_once()
sampled_b = run_once()
result = {"title": title, "valence": valence, "arousal": arousal,
          "intensity_index": intensity, "sampled": sampled_a,
          "plutchik": plutchik, "quadrant": quadrant, "seed": seed}
digest_a = hashlib.sha256(json.dumps(result, sort_keys=True).encode("utf-8")).hexdigest()
digest_b = hashlib.sha256(json.dumps(result, sort_keys=True).encode("utf-8")).hexdigest()
print("Emotion record: " + title)
print("Valence: %.2f  Arousal: %.2f" % (valence, arousal))
print("Intensity index I = 5 x (V + A) = %.2f" % intensity)
print("Sampled intensity (fixed seed %d): %.2f" % (seed, sampled_a))
print("Plutchik position for this emotion: %d of 10" % plutchik)
print("Circumplex quadrant: " + quadrant)
print("SHA-256 run 1: " + digest_a)
print("SHA-256 run 2: " + digest_b)
if digest_a == digest_b:
    print("Deterministic: MATCH - both runs produced the same intensity index.")
else:
    print("Deterministic: MISMATCH - the two runs disagreed.")
print("Educational demo only: this demo computes an intensity index from the published placement of the emotion; it describes the emotion, offers no guidance on producing it, and measures no person's feeling.")
print("Design study: this record describes a Signature design study, not an established public record.")
