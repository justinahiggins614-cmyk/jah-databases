Sunset Coast Grand Prix Rank 1 · simple · racing · beginner

Sunset Coast Grand Prix is an original racing game. A seaside grand prix winds along cliffs at dusk, where drivers race the setting sun as much as each other.

Each tick you choose accelerate, brake, or a lane change. Speed builds quickly, obstacles punish fast drivers who stay in the wrong lane, and crashes cost precious seconds. Finish the distance and your time is ranked against a set of rival times.

Watch the distance markers: an obstacle only punishes a driver who is still moving fast in its lane when the car reaches it.

Original game note: This is an original arcade-style game created for this Signature edition, with its own title, rules, and play.

How to play

Type accel, brake, left or right each tick, or quit to stop.

Working demo program

Save the code below as game.py and run python3 game.py --play to play it. Running it without arguments replays the scripted session whose output is stored with the record.

import sys
PARAMS = {'distance': 211, 'obstacles': [[44, 1], [79, 2], [114, 0], [149, 1]], 'rivals': [25, 28, 37]}
AUTO = ['accel', 'accel', 'right', 'accel', 'left', 'brake', 'accel', 'right', 'accel', 'accel', 'accel', 'left', 'right', 'accel', 'accel', 'accel', 'accel', 'brake']

def race(cmds):
    dist, speed, lane, time, penalties = 0, 0, 1, 0, 0
    obstacles = dict((d, l) for d, l in PARAMS["obstacles"])
    log = []
    for tick, cmd in enumerate(cmds):
        if cmd == "accel":
            speed = min(200, speed + 20)
        elif cmd == "brake":
            speed = max(0, speed - 40)
        elif cmd == "left":
            lane = max(0, lane - 1)
        elif cmd == "right":
            lane = min(2, lane + 1)
        dist += speed // 10
        time += 1
        crashed = False
        for d, l in obstacles.items():
            if dist >= d and dist - speed // 10 < d and l == lane and speed > 60:
                speed = 0
                penalties += 5
                crashed = True
        log.append("Tick %d: %s dist %d speed %d lane %d %s" % (tick + 1, cmd, dist, speed, lane, "CRASH" if crashed else "ok"))
        if dist >= PARAMS["distance"]:
            break
    return dist, time + penalties, log

def auto_play():
    print("Racing tribute: auto session")
    dist, total, log = race(AUTO)
    for line in log:
        print(line)
    rivals = PARAMS["rivals"]
    place = 1 + sum(1 for r in rivals if r < total)
    print("FINAL SCORE: %d" % max(0, 1000 - total * 10 + (4 - place) * 50))
    print("RESULT: FINISHED place %d time %d" % (place, total))

def play():
    print("Racing tribute: type accel/brake/left/right each tick, or quit. Finish %d m first." % PARAMS["distance"])
    cmds = []
    for line in sys.stdin:
        cmd = line.strip().lower()
        if cmd == "quit":
            break
        if cmd not in ("accel", "brake", "left", "right"):
            print("Unknown command: %s" % cmd)
            continue
        cmds.append(cmd)
        dist, total, log = race(cmds)
        print(log[-1])
        if dist >= PARAMS["distance"]:
            break
    dist, total, log = race(cmds)
    rivals = PARAMS["rivals"]
    place = 1 + sum(1 for r in rivals if r < total)
    print("FINAL SCORE: %d" % max(0, 1000 - total * 10 + (4 - place) * 50))
    print("RESULT: %s" % ("FINISHED place %d time %d" % (place, total) if dist >= PARAMS["distance"] else "QUIT"))

if __name__ == "__main__":
    if "--play" in sys.argv:
        play()
    else:
        auto_play()
