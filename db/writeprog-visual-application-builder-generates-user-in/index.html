Working implementation

function vabSpec(){return {widgets:[]};}
function vabAdd(spec,type,props){spec.widgets.push({type:type,props:props||{}});return spec;}
function vabRenderWidget(w){var p=w.props;
if(w.type==='button')return '<button id="'+p.id+'">'+p.label+'</button>';
if(w.type==='input')return '<input id="'+p.id+'" placeholder="'+(p.placeholder||'')+'">';
if(w.type==='label')return '<label>'+p.text+'</label>';
if(w.type==='list')return '<ul id="'+p.id+'">'+(p.items||[]).map(function(i){return '<li>'+i+'</li>';}).join('')+'</ul>';
return '<div>unknown</div>';}
function vabBuild(spec){var html=spec.widgets.map(vabRenderWidget).join('\n');
var js=spec.widgets.filter(function(w){return w.type==='button';}).map(function(w){return "document.getElementById('"+w.props.id+"').addEventListener('click',function(){ console.log('clicked "+w.props.id+"'); });";}).join('\n');
return '<!DOCTYPE html><html><body>\n'+html+'\n<script>\n'+js+'\n</script>\n</body></html>';}
