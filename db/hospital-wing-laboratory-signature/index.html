# Signature Truth Window — Signature design prototype
def critical_value_alert(test, value, low, high, unit):
    """Flags lab results outside critical thresholds for immediate call."""
    if value < low: return f"CRITICAL LOW: {test} {value}{unit} - CALL PROVIDER NOW"
    if value > high: return f"CRITICAL HIGH: {test} {value}{unit} - CALL PROVIDER NOW"
    return f"{test} {value}{unit} - within critical limits"
print(critical_value_alert("Potassium", 6.4, 2.8, 6.0, " mmol/L"))
