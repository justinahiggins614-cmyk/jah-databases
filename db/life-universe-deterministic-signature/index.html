import hashlib, json
title = 'The Animal Kingdom'
defining_traits = ['many cells', 'no cell walls', 'eats other living things', 'moves at some life stage']
cases = [(['many cells', 'no cell walls', 'eats other living things', 'moves at some life stage', 'sail-filterer'], True, True, True), (['many cells', 'no cell walls', 'eats other living things', 'moves at some life stage', 'roller'], True, True, True), (['many cells', 'no cell walls', 'eats other living things', 'moves at some life stage', 'colour changer'], True, True, True), (['many cells', 'no cell walls', 'eats other living things', 'moves at some life stage', 'larva swims'], True, True, True), (['many cells', 'no cell walls', 'eats other living things', 'moves at some life stage', 'filter feeder'], True, True, True), (['many cells', 'cell walls', 'makes food from sunlight'], False, False, True), (['one cell', 'no cell walls', 'eats other living things'], False, False, True), (['many cells', 'no cell walls', 'makes food from sunlight', 'moves at some life stage'], False, False, True), (['many cells', 'no cell walls', 'eats other living things'], False, False, True), (['many cells', 'no cell walls', 'eats other living things', 'moves at some life stage', 'real shark'], True, True, True)]

    return [classify(t) for (t, e, g, p) in cases]

correct = sum(1 for c in cases if c[3])
total = len(cases)

digest1 = hashlib.sha256(json.dumps(results, sort_keys=True).encode()).hexdigest()
results2 = run_once()
digest2 = hashlib.sha256(json.dumps(results2, sort_keys=True).encode()).hexdigest()

for i, (traits, exp, got, ok) in enumerate(cases, 1):

print("Accuracy: %.3f" % accuracy)
print("Baseline: %.3f" % baseline)
print("Change vs baseline: %+.3f" % (accuracy - baseline))

    print("REGRESSION FLAGGED - score fell below the stored baseline.")
else:
    print("No regression - score at or above the stored baseline.")
print("Digest run 1: " + digest1[:16])
