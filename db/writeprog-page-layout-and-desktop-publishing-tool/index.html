Working implementation

function pgPage(w,h){return {w:w,h:h,boxes:[]};}
function pgAddBox(pg,x,y,w,h,label){pg.boxes.push({x:x,y:y,w:w,h:h,label:label||''});return pg;}
function pgFlow(pg,items,margin,boxH){var x=margin,y=margin;items.forEach(function(it){if(x+it.w>pg.w-margin){x=margin;y+=boxH+margin;}pgAddBox(pg,x,y,it.w,boxH,it.label);x+=it.w+margin;});return pg;}
function pgOverlaps(pg){for(var i=0;i<pg.boxes.length;i++)for(var j=i+1;j<pg.boxes.length;j++){var a=pg.boxes[i],b=pg.boxes[j];
if(a.x<b.x+b.w&&b.x<a.x+a.w&&a.y<b.y+b.h&&b.y<a.y+a.h)return [a.label,b.label];}return null;}
function pgBounds(pg){var xs=pg.boxes.map(function(b){return b.x+b.w;});var ys=pg.boxes.map(function(b){return b.y+b.h;});
return {w:xs.length?Math.max.apply(null,xs):0,h:ys.length?Math.max.apply(null,ys):0};}
