Working implementation

function edBuf(t){return {text:t||'',cursors:[0]};}
function edAddCursor(b,pos){b.cursors.push(pos);return b;}
function edInsertAll(b,s){var off=0;var txt=b.text;b.cursors.slice().sort(function(a,c){return a-c;}).forEach(function(p){var q=p+off;txt=txt.slice(0,q)+s+txt.slice(q);off+=s.length;});b.text=txt;return b;}
function edSelect(b,a,z){return b.text.slice(a,z);}
function edUndoStack(){var s=[];return {push:function(x){s.push(x);},pop:function(){return s.pop();},depth:function(){return s.length;}};}
