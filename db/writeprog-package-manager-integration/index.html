Working implementation

function pmInstallOrder(pkgs){var order=[],done={},visiting={};
function visit(n){if(done[n])return;if(visiting[n])throw new Error('cycle '+n);visiting[n]=1;
var p=pkgs[n];if(!p)throw new Error('unknown package '+n);(p.deps||[]).forEach(visit);visiting[n]=0;done[n]=1;order.push(n+'@'+p.version);}
Object.keys(pkgs).forEach(visit);return order;}
function pmLock(pkgs){var lock={};Object.keys(pkgs).forEach(function(n){lock[n]=pkgs[n].version;});return lock;}
