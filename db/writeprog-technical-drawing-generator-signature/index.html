Working implementation

function tdNew(){return {shapes:[]};}
function tdRect(d,x,y,w,h){d.shapes.push({t:'rect',x:x,y:y,w:w,h:h});return d;}
function tdCircle(d,cx,cy,r){d.shapes.push({t:'circle',cx:cx,cy:cy,r:r});return d;}
function tdDim(d,x1,y1,x2,y2,label){d.shapes.push({t:'dim',x1:x1,y1:y1,x2:x2,y2:y2,label:label});return d;}
function tdSvg(d){var p=['<svg xmlns="http://www.w3.org/2000/svg" width="400" height="300">'];
d.shapes.forEach(function(s){if(s.t==='rect')p.push('<rect x="'+s.x+'" y="'+s.y+'" width="'+s.w+'" height="'+s.h+'" fill="none" stroke="black"/>');
else if(s.t==='circle')p.push('<circle cx="'+s.cx+'" cy="'+s.cy+'" r="'+s.r+'" fill="none" stroke="black"/>');
else if(s.t==='dim')p.push('<line x1="'+s.x1+'" y1="'+s.y1+'" x2="'+s.x2+'" y2="'+s.y2+'" stroke="red"/><text x="'+((s.x1+s.x2)/2)+'" y="'+((s.y1+s.y2)/2-4)+'">'+s.label+'</text>');});
p.push('</svg>');return p.join('');}
function tdArea(d){return d.shapes.reduce(function(a,s){return a+(s.t==='rect'?s.w*s.h:s.t==='circle'?Math.PI*s.r*s.r:0);},0);}
