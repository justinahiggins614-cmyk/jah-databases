design_title = 'The Digital Pigeon Parliament Field Guide Index'
design = 'A pigeon parliament website: a visitor picks a motion, such as where the flock meets or which rooftop is best, and the parliament shows the motion in plain words, lets every pigeon member vote once, and writes the result where everyone can see it. Every motion is a playful design study, and the site never claims real pigeons vote or real laws are made. This record is the field guide index for the whole database: it lists all twenty-three content records in ladder order, simple to compounded to complex, and tells a reader where to start.'
steps = ['Add a tournament bracket with rounds, divisions, and a season', 'Add a creator studio so visitors can build a new parliament session', 'Add a review step so every new parliament session is checked before it is published', 'Add the parliament session in many languages, with the same rules in each', 'Add an offline pack so the parliament session works with no connection']
count = 23
each = 1
total = count * each
print("Website design study: " + design_title)
print("Design (a design study, not a live website): " + design)
print("Note: this program simulates the design on paper; it is not a test on a live website.")
for i, step in enumerate(steps, 1):
    print("Step %d: %s" % (i, step))
print("Count: %d" % count)
print("Each: %d" % each)
print("Total: %d" % total)
print("Feasibility: %s" % 'buildable-now')
