Working implementation

function shingles(text,n){n=n||3;var ws=text.toLowerCase().match(/[a-z0-9]+/g)||[];var s={};for(var i=0;i+n<=ws.length;i++)s[ws.slice(i,i+n).join(' ')]=1;return s;}
function jaccard(a,b){var A=shingles(a),B=shingles(b);var inter=0,union=0;var seen={};
Object.keys(A).forEach(function(k){seen[k]=1;if(B[k])inter++;});Object.keys(B).forEach(function(k){if(!seen[k])seen[k]=1;});
union=Object.keys(seen).length;return union===0?1:inter/union;}
function similarityReport(a,b){var s=jaccard(a,b);return {similarity:+s.toFixed(3),verdict:s>0.8?'likely duplicate':s>0.4?'suspicious overlap':'original'};}
