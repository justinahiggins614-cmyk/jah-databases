Neon Loom Rank 1 · simple · vector graphics · beginner

A weaver pilots a shuttle through a glowing loom of light-threads, cutting stray threads that tangle into knots. The shuttle drifts with momentum, and each cut thread unravels a knot worth points.

The rules are simple: the ship drifts with momentum on a 20 by 16 grid, thrusting, turning, and firing at 5 drifting rocks. Large rocks split into smaller ones when shot, and the run ends when every rock is cleared or the ship collides three times.

Knots drift and split when cut, so the weaver cuts the largest knot first and lets fragments drift apart before chasing them. A clean cut scores double, while a tangled shuttle loses a thread of its own.

Original game note: This is an original arcade-style game created for this Signature edition, with its own title, rules, and play.

How to play

Type thrust/left/right/shoot to fly and fire, quit to stop.

import sys
PARAMS = {'w': 20, 'h': 16, 'ship': [5, 4], 'rocks': [[2, 3, -1, 1, 2], [6, 5, 1, 0, 1], [10, 7, -1, 1, 1], [14, 9, 1, 0, 1], [3, 11, -1, 1, 2]]}
AUTO = ['thrust', 'shoot', 'left', 'thrust', 'shoot', 'right', 'right', 'thrust', 'shoot', 'left', 'shoot', 'thrust', 'thrust', 'shoot', 'right', 'shoot', 'left', 'thrust']

DIRS_V = [(1, 0), (0, 1), (-1, 0), (0, -1)]

def auto_play():
    W, H = PARAMS["w"], PARAMS["h"]
    ship = list(PARAMS["ship"])
    heading = 0
    rocks = [list(r) for r in PARAMS["rocks"]]
    score, lives = 0, 3
    print("Vector tribute: auto session")
    for i, cmd in enumerate(AUTO):
        if cmd == "left":
            heading = (heading - 1) % 4
        elif cmd == "right":
            heading = (heading + 1) % 4
        elif cmd == "thrust":
            dx, dy = DIRS_V[heading]
            ship[0] = (ship[0] + dx * 2) % W
            ship[1] = (ship[1] + dy * 2) % H
        elif cmd == "shoot":
            dx, dy = DIRS_V[heading]
            for k in range(1, 4):
                target = ((ship[0] + dx * k) % W, (ship[1] + dy * k) % H)
                hit = None
                for rock in rocks:
                    if (rock[0], rock[1]) == target:
                        hit = rock
                        break
                if hit is not None:
                    rocks.remove(hit)
                    score += hit[4] * 20
                    if hit[4] == 2:
                        rocks.append([hit[0], hit[1], 1, 0, 1])
                        rocks.append([hit[0], hit[1], 0, 1, 1])
                    break
        new_rocks = []
        for rock in rocks:
            rock[0] = (rock[0] + rock[2]) % W
            rock[1] = (rock[1] + rock[3]) % H
            new_rocks.append(rock)
        rocks = new_rocks
        if any((rock[0], rock[1]) == (ship[0], ship[1]) for rock in rocks):
            lives -= 1
            rocks = [r for r in rocks if (r[0], r[1]) != (ship[0], ship[1])]
        print("Step %d: %s ship %s rocks %d score %d lives %d" % (i + 1, cmd, tuple(ship), len(rocks), score, lives))
        if not rocks or lives <= 0:
            break
    print("FINAL SCORE: %d" % score)
    print("RESULT: %s" % ("WIN" if not rocks else "OUT OF LIVES" if lives <= 0 else "ONGOING"))

The rules are simple: the ship drifts with momentum on a 20 by 16 grid, thrusting, turning, and firing at 6 drifting rocks. Large rocks split into smaller ones when shot, and the run ends when every rock is cleared or the ship collides three times.

Angle-sharks circle islands in pairs, so the pilots secure islands one at a time, clearing both sharks before landing. A bridged island pair doubles the score of the next island secured.

import sys
PARAMS = {'w': 20, 'h': 16, 'ship': [10, 7], 'rocks': [[13, 5, -1, 1, 2], [2, 7, 1, 0, 1], [6, 9, -1, 1, 1], [10, 11, 1, 0, 1], [14, 13, -1, 1, 2], [3, 4, 1, 0, 1]]}
AUTO = ['shoot', 'thrust', 'thrust', 'right', 'shoot', 'left', 'thrust', 'shoot', 'shoot', 'right', 'thrust', 'left', 'shoot', 'thrust', 'shoot']

DIRS_V = [(1, 0), (0, 1), (-1, 0), (0, -1)]

def auto_play():
    W, H = PARAMS["w"], PARAMS["h"]
    ship = list(PARAMS["ship"])
    heading = 0
    rocks = [list(r) for r in PARAMS["rocks"]]
    score, lives = 0, 3
    print("Vector tribute: auto session")
    for i, cmd in enumerate(AUTO):
        if cmd == "left":
            heading = (heading - 1) % 4
        elif cmd == "right":
            heading = (heading + 1) % 4
        elif cmd == "thrust":
            dx, dy = DIRS_V[heading]
            ship[0] = (ship[0] + dx * 2) % W
            ship[1] = (ship[1] + dy * 2) % H
        elif cmd == "shoot":
            dx, dy = DIRS_V[heading]
            for k in range(1, 4):
                target = ((ship[0] + dx * k) % W, (ship[1] + dy * k) % H)
                hit = None
                for rock in rocks:
                    if (rock[0], rock[1]) == target:
                        hit = rock
                        break
                if hit is not None:
                    rocks.remove(hit)
                    score += hit[4] * 20
                    if hit[4] == 2:
                        rocks.append([hit[0], hit[1], 1, 0, 1])
                        rocks.append([hit[0], hit[1], 0, 1, 1])
                    break
        new_rocks = []
        for rock in rocks:
            rock[0] = (rock[0] + rock[2]) % W
            rock[1] = (rock[1] + rock[3]) % H
            new_rocks.append(rock)
        rocks = new_rocks
        if any((rock[0], rock[1]) == (ship[0], ship[1]) for rock in rocks):
            lives -= 1
            rocks = [r for r in rocks if (r[0], r[1]) != (ship[0], ship[1])]
        print("Step %d: %s ship %s rocks %d score %d lives %d" % (i + 1, cmd, tuple(ship), len(rocks), score, lives))
        if not rocks or lives <= 0:
            break
    print("FINAL SCORE: %d" % score)
    print("RESULT: %s" % ("WIN" if not rocks else "OUT OF LIVES" if lives <= 0 else "ONGOING"))
