def colonoscopy_due(age, last_result, years_since):
    """Simplified colonoscopy screening scheduler (average risk)."""
    if age < 45: return "Routine screening starts at 45"
    interval = 10 if last_result == "normal" else 5 if last_result == "small polyps" else 3
    due = interval - years_since
