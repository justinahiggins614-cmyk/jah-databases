The UNESCO World Heritage List is the register of the world's most outstanding places, and it is the list every other heritage register measures itself against. import hashlib
import json
import random

title = 'The GeoNames Landmark Gazetteer'
points = [('Toponyms in the gazetteer, millions', 12), ('Year GeoNames was founded', 2005), ('Feature code classes (spot, parks, mountains, and others)', 9), ('Years the gazetteer has been growing, approx.', 21)]
seed = 2005
claimed_note = 'As reported: the GeoNames gazetteer holds more than 12 million toponyms, published at geonames.org under CC-BY 4.0.'

def compute():
    rng = random.Random(seed)
    sample = rng.choice([p[0] for p in points])
    values = [p[1] for p in points]
    total = sum(values)
    mean = total / len(values)
    lo = min(values)
    hi = max(values)
    ratio = hi / lo if lo else 0
    return {"title": title, "points": len(points), "total": total,
            "mean": round(mean, 3), "min": lo, "max": hi,
            "max_min_ratio": round(ratio, 3), "spot_check": sample}

result_a = compute()
result_b = compute()
digest_a = hashlib.sha256(json.dumps(result_a, sort_keys=True).encode("utf-8")).hexdigest()
digest_b = hashlib.sha256(json.dumps(result_b, sort_keys=True).encode("utf-8")).hexdigest()
print("Record figures check: " + title)
for label, value in points:
    print("  %s: %s" % (label, value))
print("Points: %d  Total: %s  Mean: %.3f  Min: %s  Max: %s  Max/min ratio: %.3f" % (result_a["points"], result_a["total"], result_a["mean"], result_a["min"], result_a["max"], result_a["max_min_ratio"]))
print("Spot check (seeded pick): " + result_a["spot_check"])
print("Claimed result as reported: " + claimed_note)
print("SHA-256 run 1: " + digest_a)
print("SHA-256 run 2: " + digest_b)
if digest_a == digest_b:
    print("Deterministic: MATCH - both runs produced the same figures check.")
else:
    print("Deterministic: MISMATCH - the two runs disagreed.")
print("Note: this demo recomputes descriptive statistics from the record's own published figures. It does not re-derive the original graph or database, and it proves nothing beyond the arithmetic shown.")
