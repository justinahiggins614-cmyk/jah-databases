def sieve(limit):
    keep = [True] * limit
    keep[0] = keep[1] = False
    for i in range(2, limit):
        if keep[i]:
            for j in range(i * i, limit, i):
                keep[j] = False
    return [n for n in range(limit) if keep[n]]
