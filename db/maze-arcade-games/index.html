Pac-Man Rank 1 · simple · maze · beginner

Pac-Man arrived in 1980 and changed what an arcade game could be about. Instead of shooting, the player steered a round hero through a maze, eating dots while four ghosts with distinct personalities patrolled the corridors. Large power pellets briefly let the hero turn the tables and hunt the ghosts, and the game's cheerful look drew in players who had never touched a shooter.

This tribute recreates the maze chase with a winding corridor map, a full field of dots, and a single pursuing ghost. The demo changes the original by shrinking the ghost cast to one pursuer and scripting a clean route through the maze, so the dot-eating rhythm is easy to follow. It is an original tribute inspired by the mechanics, not the game itself.

Maze runners learn the map before they chase the score. Clear the edges early, memorize the escape routes from every corner, and never enter a corridor you cannot leave. The players who lasted longest treated the maze as a place they lived in, not a puzzle they solved once.

Playable tribute note: The demo below is an original tribute inspired by the mechanics of Pac-Man (1980). It is not Pac-Man (1980), and it uses no art, music, or code from that game.

import sys
PARAMS = {'map': ['###########', '#P........#', '#....#....#', '#....#....#', '#..#......#', '#........G#', '###########']}
AUTO = ['right', 'right', 'right', 'right', 'down', 'down', 'right', 'right', 'down', 'left', 'left']

def parse_map():
    walls, dots, player, ghost = set(), set(), None, None
    for y, row in enumerate(PARAMS["map"]):
        for x, ch in enumerate(row):
            if ch == "#":
                walls.add((x, y))
            elif ch == ".":
                dots.add((x, y))
            elif ch == "P":
                player = (x, y)
            elif ch == "G":
                ghost = (x, y)
    return walls, dots, player, ghost

DIRS = {"up": (0, -1), "down": (0, 1), "left": (-1, 0), "right": (1, 0)}

def step(state, cmd):
    walls, dots, player, ghost, score, lives = state
    dx, dy = DIRS.get(cmd, (0, 0))
    nx, ny = player[0] + dx, player[1] + dy
    if (nx, ny) not in walls:
        player = (nx, ny)
    if player in dots:
        dots = dots - {player}
        score += 10
    if ghost is not None:
        gx, gy = ghost
        best, bestd = ghost, abs(gx - player[0]) + abs(gy - player[1])
        for mx, my in [(1, 0), (-1, 0), (0, 1), (0, -1)]:
            cand = (gx + mx, gy + my)
            if cand in walls:
                continue
            d = abs(cand[0] - player[0]) + abs(cand[1] - player[1])
            if d < bestd:
                best, bestd = cand, d
        ghost = best
    if ghost is not None and player == ghost:
        lives -= 1
        walls, dots, player, ghost = reset_positions()
    return (walls, dots, player, ghost, score, lives)

def reset_positions():
    walls, dots, player, ghost = parse_map()
    return walls, dots, player, ghost

def result_of(state):
    walls, dots, player, ghost, score, lives = state
    if not dots:
        return "WIN"
    if lives <= 0:
        return "OUT OF LIVES"
    return "ONGOING"

def auto_play():
    walls, dots, player, ghost = parse_map()
    state = (walls, dots, player, ghost, 0, 3)
    print("Maze tribute: auto session")
    for i, cmd in enumerate(AUTO):
        state = step(state, cmd)
        walls, dots, player, ghost, score, lives = state
        print("Step %d: %s -> pos %s score %d dots %d lives %d" % (i + 1, cmd, player, score, len(dots), lives))
        if result_of(state) != "ONGOING":
            break
    print("FINAL SCORE: %d" % state[4])
    print("RESULT: %s" % result_of(state))

def play():
    walls, dots, player, ghost = parse_map()
    state = (walls, dots, player, ghost, 0, 3)
    print("Maze tribute: type up/down/left/right, or quit. Eat every dot, avoid the ghost.")
    for line in sys.stdin:
        cmd = line.strip().lower()
        if cmd == "quit":
            break
        if cmd not in DIRS:
            print("Unknown command: %s" % cmd)
            continue
        state = step(state, cmd)
        walls, dots, player, ghost, score, lives = state
        print("pos %s score %d dots %d lives %d" % (player, score, len(dots), lives))
        if result_of(state) != "ONGOING":
            break
    print("FINAL SCORE: %d" % state[4])
    print("RESULT: %s" % result_of(state))

if __name__ == "__main__":
    if "--play" in sys.argv:
        play()
    else:
        auto_play()
