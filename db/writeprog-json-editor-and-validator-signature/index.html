Working implementation

function jsonParse(s){try{return {ok:true,value:JSON.parse(s)};}catch(e){return {ok:false,error:String(e.message)};}}
function jsonValidateSchema(val,schema){var errs=[];
function chk(v,s,path){if(s.type==='object'){if(typeof v!=='object'||v===null||Array.isArray(v)){errs.push(path+': expected object');return;}
(s.required||[]).forEach(function(k){if(!(k in v))errs.push(path+'.'+k+': required');});
Object.keys(s.properties||{}).forEach(function(k){if(k in v)chk(v[k],s.properties[k],path+'.'+k);});}
else if(s.type==='array'){if(!Array.isArray(v)){errs.push(path+': expected array');return;}v.forEach(function(x,i){chk(x,s.items||{},path+'['+i+']');});}
else if(s.type==='string'){if(typeof v!=='string')errs.push(path+': expected string');}
else if(s.type==='number'){if(typeof v!=='number')errs.push(path+': expected number');}
else if(s.type==='boolean'){if(typeof v!=='boolean')errs.push(path+': expected boolean');}}
chk(val,schema,'$');return {ok:errs.length===0,errors:errs};}
function jsonPretty(s){var r=jsonParse(s);return r.ok?JSON.stringify(r.value,null,2):null;}
