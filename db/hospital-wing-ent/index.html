def hearing_loss_class(db_loss):
    """Classifies hearing loss severity from pure-tone average dB HL."""
    if db_loss <= 25: return "Normal hearing"
    if db_loss <= 40: return "Mild loss - consider hearing aids"
    if db_loss <= 70: return "Moderate-severe loss - hearing aids recommended"
    if db_loss <= 90: return "Severe loss - powerful aids or implant eval"
    return "Profound loss - cochlear implant evaluation"
