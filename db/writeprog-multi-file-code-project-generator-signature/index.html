Working implementation

function scaffoldProject(name,opts){opts=opts||{};var files={};
files['package.json']=JSON.stringify({name:name,version:'1.0.0',main:'src/index.js'},null,2);
files['src/index.js']="'use strict';\nconsole.log('"+name+" running');\n";
files['README.md']='# '+name+'\n\nGenerated project scaffold.\n';
if(opts.test)files['test/test.js']="const assert=require('assert');\ndescribe('smoke',()=>{it('works',()=>assert.ok(true));});\n";
if(opts.cli)files['bin/cli.js']="#!/usr/bin/env node\nrequire('../src/index.js');\n";
return {name:name,files:files};}
function scaffoldList(proj){return Object.keys(proj.files);}
