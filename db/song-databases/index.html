Science proof

A catalog spot-check works like a quality audit: a fixed-seed sample of entries is drawn from the catalog's categories, the counts per category are compared with the flat expectation, and a chi-square figure measures how far the sample strays from flat. The demo runs that audit twice with the same seed and hashes both results, so a fresh run verifies the arithmetic exactly. The audit proves only that the demo's own arithmetic is deterministic; it does not query the live database and it proves nothing about the live holdings, which must be checked against the published source itself. The science of MusicBrainz is the science of identification: a recording must be told apart from every other recording of the same song, and the database does that with unique identifiers, fingerprints, and a public edit history. In plain words, the demo draws 25 entries at random with a fixed seed across 4 categories, so each category should receive about 6.25 entries by chance alone. The demo's simulated run gave counts {"artists": 7, "recordings": 4, "releases": 8, "works": 6} and a chi-square of 1.400. A chi-square near zero is exactly what a flat random draw produces; a large chi-square would need a reason, and the audit proves only that the demo's own arithmetic is deterministic, not that the live database is balanced.

Working demo program

import hashlib
import json
import math
import random

title = 'MusicBrainz'
entries = 2000000
categories = ['artists', 'releases', 'recordings', 'works']
trials = 25
seed = 2000
claimed_note = 'As published: millions of releases and tens of millions of recordings, CC0 data, free with a public API and bulk downloads.'

def run_once():
    rng = random.Random(seed)
    counts = {c: 0 for c in categories}
    for _ in range(trials):
        counts[rng.choice(categories)] += 1
    return counts

expected = trials / len(categories)
counts_a = run_once()
counts_b = run_once()
chi2 = sum((o - expected) ** 2 / expected for o in counts_a.values())
result = {"title": title, "entries": entries, "trials": trials,
          "categories": len(categories), "seed": seed,
          "counts": counts_a, "chi_square": round(chi2, 3)}
digest_a = hashlib.sha256(json.dumps(result, sort_keys=True).encode("utf-8")).hexdigest()
digest_b = hashlib.sha256(json.dumps(result, sort_keys=True).encode("utf-8")).hexdigest()
print("Catalog: " + title)
print("Entries (published, approximate): %d" % entries)
print("Spot-check sample: %d entries across %d categories (fixed seed %d)" % (trials, len(categories), seed))
print("Category counts: " + json.dumps(counts_a, sort_keys=True))
print("Expected per category: %.2f entries" % expected)
print("Chi-square: %.3f (sum of (observed minus expected) squared, divided by expected)" % chi2)
print("Spot-check note: " + claimed_note)
print("SHA-256 run 1: " + digest_a)
print("SHA-256 run 2: " + digest_b)
if digest_a == digest_b:
    print("Deterministic: MATCH - both runs produced the same sampled counts.")
else:
    print("Deterministic: MISMATCH - the two runs disagreed.")
print("Note: this demo simulates a seeded spot-check sample only. It does not query the live database, and a balanced sample is chance, not a verification of the live holdings.")

import json
import math
import random

title = 'The Internet Archive Audio Archive'

    counts = {c: 0 for c in categories}

    return counts

expected = trials / len(categories)

counts_b = run_once()

import json
import math
import random

title = 'WhoSampled'
entries = 500000
categories = ['samples', 'covers', 'remixes', 'artists']
trials = 25
seed = 2169
claimed_note = 'As published: hundreds of thousands of sampled tracks and sample connections, free to browse on the web.'

def run_once():
    rng = random.Random(seed)
    counts = {c: 0 for c in categories}
    for _ in range(trials):
        counts[rng.choice(categories)] += 1
    return counts

expected = trials / len(categories)
counts_a = run_once()
counts_b = run_once()
chi2 = sum((o - expected) ** 2 / expected for o in counts_a.values())
result = {"title": title, "entries": entries, "trials": trials,
          "categories": len(categories), "seed": seed,
          "counts": counts_a, "chi_square": round(chi2, 3)}

import json
import math
import random

categories = ['tracks', 'artists', 'playlists', 'comments']

    return counts

expected = trials / len(categories)

counts_b = run_once()

import math
import random

    return counts

counts_b = run_once()

Working demo program

import hashlib
import json
import math
import random

title = 'The Song Databases Field Guide Index'
entries = 24
categories = ['metadata databases', 'lyrics databases', 'listening databases', 'guides']
trials = 25
seed = 2299
claimed_note = 'As published in this pair: a complete list of the twenty-three song databases in this pair, with the question each one answers.'

def run_once():
    rng = random.Random(seed)
    counts = {c: 0 for c in categories}
    for _ in range(trials):
        counts[rng.choice(categories)] += 1
    return counts

expected = trials / len(categories)
counts_a = run_once()
counts_b = run_once()
chi2 = sum((o - expected) ** 2 / expected for o in counts_a.values())
result = {"title": title, "entries": entries, "trials": trials,
          "categories": len(categories), "seed": seed,
          "counts": counts_a, "chi_square": round(chi2, 3)}
digest_a = hashlib.sha256(json.dumps(result, sort_keys=True).encode("utf-8")).hexdigest()
digest_b = hashlib.sha256(json.dumps(result, sort_keys=True).encode("utf-8")).hexdigest()
print("Catalog: " + title)
print("Entries (published, approximate): %d" % entries)
print("Spot-check sample: %d entries across %d categories (fixed seed %d)" % (trials, len(categories), seed))
print("Category counts: " + json.dumps(counts_a, sort_keys=True))
print("Expected per category: %.2f entries" % expected)
print("Chi-square: %.3f (sum of (observed minus expected) squared, divided by expected)" % chi2)
print("Spot-check note: " + claimed_note)
print("SHA-256 run 1: " + digest_a)
print("SHA-256 run 2: " + digest_b)
if digest_a == digest_b:
    print("Deterministic: MATCH - both runs produced the same sampled counts.")
else:
    print("Deterministic: MISMATCH - the two runs disagreed.")
print("Note: this demo simulates a seeded spot-check sample only. It does not query the live database, and a balanced sample is chance, not a verification of the live holdings.")
