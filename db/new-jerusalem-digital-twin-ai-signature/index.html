def main():
    refs = [0.0, 10.0, 20.0, 30.0]
    readings = [0.2, 10.1, 20.3, 29.8]
    n = len(refs)
    mx = sum(refs) / n
    my = sum(readings) / n
    slope = sum((x - mx) * (y - my) for x, y in zip(refs, readings)) / sum((x - mx) ** 2 for x in refs)
    intercept = my - slope * mx
    print(f'slope: {slope:.4f}')
    print(f'intercept: {intercept:.4f}')
    print(f'max residual: {max(abs(y - (slope * x + intercept)) for x, y in zip(refs, readings)):.3f}')
