Iron Fist of the Harbor Rank 1 · simple · fighting · beginner

Iron Fist of the Harbor is an original fighting game. Dockworkers, sailors, and traveling fighters gather at a harbor festival where bouts are fought on a wooden pier stage, and the crowd cheers every clean hit.

Bouts are fought in rounds: jab is quick but light, hook hits harder, kick hits hardest of the basic moves, special delivers a big blow but needs a rest before it can be used again, and block cuts incoming damage sharply. The fighter who empties the opponent's health first wins the bout.

Players who win consistently mix fast jabs with occasional kicks, keeping the opponent guessing instead of repeating one move.

Original game note: This is an original arcade-style game created for this Signature edition, with its own title, rules, and play.

How to play

Type jab, hook, kick, special or block each round, or quit to stop.

Working demo program

Save the code below as game.py and run python3 game.py --play to play it. Running it without arguments replays the scripted session whose output is stored with the record.

import sys
PARAMS = {'cpu': ['hook', 'kick', 'block', 'jab']}
AUTO = ['jab', 'hook', 'kick', 'special', 'block', 'jab', 'hook', 'kick', 'jab', 'hook']

MOVES = {"jab": 6, "hook": 10, "kick": 14, "special": 20, "block": 0}

def bout(moves):
    php, chp = 100, 100
    cpu_pattern = PARAMS["cpu"]
    special_cd = 0
    log = []
    for rnd, mv in enumerate(moves):
        if mv not in MOVES:
            log.append("Round %d: unknown move %s, skipped" % (rnd + 1, mv))
            continue
        cpu = cpu_pattern[rnd % len(cpu_pattern)]
        dmg_to_cpu, dmg_to_player = 0, 0
        if mv == "special" and special_cd > 0:
            log.append("Round %d: special on cooldown, jab instead" % (rnd + 1))
            mv = "jab"
        if mv == "special":
            special_cd = 4
        if special_cd > 0:
            special_cd -= 1
        if mv != "block":
            dmg_to_cpu = MOVES[mv]
            if cpu == "block":
                dmg_to_cpu = dmg_to_cpu * 3 // 10
        if cpu != "block":
            dmg_to_player = MOVES[cpu]
            if mv == "block":
                dmg_to_player = dmg_to_player * 3 // 10
        chp = max(0, chp - dmg_to_cpu)
        php = max(0, php - dmg_to_player)
        log.append("Round %d: you %s vs cpu %s -> you HP %d cpu HP %d" % (rnd + 1, mv, cpu, php, chp))
        if php <= 0 or chp <= 0:
            break
    return php, chp, log

def auto_play():
    print("Fighting tribute: auto session")
    php, chp, log = bout(AUTO)
    for line in log:
        print(line)
    score = max(0, php - chp) + (100 if chp <= 0 else 0)
    print("FINAL SCORE: %d" % score)
    print("RESULT: %s" % ("WIN" if chp <= 0 else "LOSE" if php <= 0 else "DRAW"))

def play():
    print("Fighting tribute: type jab/hook/kick/special/block each round, or quit.")
    moves = []
    for line in sys.stdin:
        mv = line.strip().lower()
        if mv == "quit":
            break
        if mv not in MOVES:
            print("Unknown move: %s" % mv)
            continue
        moves.append(mv)
        php, chp, log = bout(moves)
        print(log[-1])
        if php <= 0 or chp <= 0:
            break
    php, chp, log = bout(moves)
    score = max(0, php - chp) + (100 if chp <= 0 else 0)
    print("FINAL SCORE: %d" % score)
    print("RESULT: %s" % ("WIN" if chp <= 0 else "LOSE" if php <= 0 else "QUIT" if not moves else "DRAW"))

if __name__ == "__main__":
    if "--play" in sys.argv:
        play()
    else:
        auto_play()

AUTO = ['jab', 'block', 'jab', 'block', 'jab', 'block', 'jab', 'block', 'jab', 'block', 'jab', 'block', 'jab', 'block', 'jab', 'block', 'jab', 'block', 'jab', 'block', 'jab', 'block']

MOVES = {"jab": 6, "hook": 10, "kick": 14, "special": 20, "block": 0}

def bout(moves):
    php, chp = 100, 100
    cpu_pattern = PARAMS["cpu"]
    special_cd = 0
    log = []
    for rnd, mv in enumerate(moves):
        if mv not in MOVES:
            log.append("Round %d: unknown move %s, skipped" % (rnd + 1, mv))
            continue
        cpu = cpu_pattern[rnd % len(cpu_pattern)]
        dmg_to_cpu, dmg_to_player = 0, 0
        if mv == "special" and special_cd > 0:
            log.append("Round %d: special on cooldown, jab instead" % (rnd + 1))
            mv = "jab"
        if mv == "special":
            special_cd = 4
        if special_cd > 0:
            special_cd -= 1
        if mv != "block":
            dmg_to_cpu = MOVES[mv]
            if cpu == "block":
                dmg_to_cpu = dmg_to_cpu * 3 // 10
        if cpu != "block":
            dmg_to_player = MOVES[cpu]
            if mv == "block":
                dmg_to_player = dmg_to_player * 3 // 10
        chp = max(0, chp - dmg_to_cpu)
        php = max(0, php - dmg_to_player)
        log.append("Round %d: you %s vs cpu %s -> you HP %d cpu HP %d" % (rnd + 1, mv, cpu, php, chp))
        if php <= 0 or chp <= 0:
            break
    return php, chp, log

def auto_play():
    print("Fighting tribute: auto session")
    php, chp, log = bout(AUTO)
    for line in log:
        print(line)
    score = max(0, php - chp) + (100 if chp <= 0 else 0)
    print("FINAL SCORE: %d" % score)
    print("RESULT: %s" % ("WIN" if chp <= 0 else "LOSE" if php <= 0 else "DRAW"))

def play():
    print("Fighting tribute: type jab/hook/kick/special/block each round, or quit.")
    moves = []
    for line in sys.stdin:

        if mv == "quit":
            break
        if mv not in MOVES:
            print("Unknown move: %s" % mv)
            continue
        moves.append(mv)
        php, chp, log = bout(moves)
        print(log[-1])
        if php <= 0 or chp <= 0:
            break
    php, chp, log = bout(moves)
    score = max(0, php - chp) + (100 if chp <= 0 else 0)
    print("FINAL SCORE: %d" % score)
    print("RESULT: %s" % ("WIN" if chp <= 0 else "LOSE" if php <= 0 else "QUIT" if not moves else "DRAW"))
