Working implementation

function dbDoc(schema){var out=['# '+schema.name+'\n','_'+(schema.description||'Database schema documentation.')+'_\n','## Tables\n'];
schema.tables.forEach(function(t){out.push('### '+t.name+'\n');out.push('| Column | Type | Constraints |');
out.push('|---|---|---|');t.columns.forEach(function(c){var cs=[];if(c.pk)cs.push('PK');if(c.notNull)cs.push('NOT NULL');if(c.unique)cs.push('UNIQUE');out.push('| '+c.name+' | '+c.type+' | '+cs.join(', ')+' |');});out.push('');});
return out.join('\n');}
function dbDocStats(schema){var cols=0;schema.tables.forEach(function(t){cols+=t.columns.length;});return {tables:schema.tables.length,columns:cols};}
