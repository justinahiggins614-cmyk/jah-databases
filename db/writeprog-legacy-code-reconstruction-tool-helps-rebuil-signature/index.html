function lcrObserve(cases){var table={};cases.forEach(function(c){table[c.input]=c.output;});return table;}
function lcrInfer(table){var keys=Object.keys(table);var nums=keys.every(function(k){return !isNaN(+k)&&!isNaN(+table[k]);});
if(nums){var pairs=keys.map(function(k){return [+k,+table[k]];});
var diffs=pairs.map(function(p){return p[1]-p[0];});var same=diffs.every(function(d){return d===diffs[0];});
if(same)return {kind:'add',by:diffs[0]};
var ratios=pairs.map(function(p){return p[1]/p[0];});if(ratios.every(function(r){return r===ratios[0];}))return {kind:'mul',by:ratios[0]};}
return {kind:'lookup'};}
function lcrRebuild(table,rule){if(rule.kind==='add')return function(x){return x+rule.by;};
if(rule.kind==='mul')return function(x){return x*rule.by;};return function(x){return table[x];};}
function lcrVerify(fn,cases){return cases.every(function(c){return fn(c.input)===c.output;});}
