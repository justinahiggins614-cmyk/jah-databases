Working demo program

import math
TITLE = 'Jasper Foundation Stone'
N = 1.52
D = 0.01
INC = 45.0
nb = N + D / 2.0
nr = N - D / 2.0
ab = math.degrees(math.asin(math.sin(math.radians(INC)) / nb))
ar = math.degrees(math.asin(math.sin(math.radians(INC)) / nr))
print(TITLE + ' - dispersion calculator')
print('Mean refractive index: %.3f, dispersion: %.3f' % (N, D))
print('Blue-side refracted angle: %.2f degrees' % ab)
print('Red-side refracted angle: %.2f degrees' % ar)
print('Angular spread: %.2f degrees' % abs(ar - ab))
