Working implementation

function dagRun(tasks){var order=[],done={},visiting={};
function visit(name){if(done[name])return;if(visiting[name])throw new Error('cycle at '+name);visiting[name]=1;
var t=tasks[name];if(!t)throw new Error('missing task '+name);(t.deps||[]).forEach(visit);visiting[name]=0;done[name]=1;order.push(name);}
Object.keys(tasks).forEach(visit);var results={};order.forEach(function(n){results[n]=tasks[n].run(results);});return {order:order,results:results};}
