function dcTrim(rows){return rows.map(function(r){var o={};Object.keys(r).forEach(function(k){o[k]=typeof r[k]==='string'?r[k].trim():r[k];});return o;});}
function dcDropEmpty(rows){return rows.filter(function(r){return Object.keys(r).some(function(k){var v=r[k];return v!==''&&v!==null&&v!==undefined;});});}
function dcDedupe(rows,key){var seen={},out=[];rows.forEach(function(r){var k=key?r[key]:JSON.stringify(r);if(!seen[k]){seen[k]=1;out.push(r);}});return out;}
function dcNormalize(rows,spec){return rows.map(function(r){var o=Object.assign({},r);Object.keys(spec).forEach(function(k){if(k in o)o[k]=spec[k](o[k]);});return o;});}
function dcPipeline(rows,steps){return steps.reduce(function(r,s){return s(r);},rows);}
