function miniCompiler(lang){ // lang: {fn: 'main'|'Main'|'run', wrap: regex pieces}
function lex(s){var toks=[],i=0;while(i<s.length){var c=s[i];if(/\s/.test(c)){i++;continue;}
if(/[0-9]/.test(c)){var j=i;while(j<s.length&&/[0-9]/.test(s[j]))j++;toks.push({t:'n',v:parseInt(s.slice(i,j),10)});i=j;continue;}
if('+-*/%()'.indexOf(c)>=0){toks.push({t:'o',v:c});i++;continue;}
if(c===';'){toks.push({t:'s'});i++;continue;}
if(/[A-Za-z_]/.test(c)){var j=i;while(j<s.length&&/[A-Za-z0-9_]/.test(s[j]))j++;toks.push({t:'w',v:s.slice(i,j)});i=j;continue;}
throw new Error(lang.name+': bad char '+c);}return toks;}
function parse(toks){var pos=0;function pk(){return toks[pos];}function nx(){return toks[pos++];}
function ex(){var l=tm();while(pk()&&pk().t==='o'&&(pk().v==='+'||pk().v==='-')){var o=nx().v;l={op:o,l:l,r:tm()};}return l;}
function tm(){var l=un();while(pk()&&pk().t==='o'&&(pk().v==='*'||pk().v==='/'||pk().v==='%')){var o=nx().v;l={op:o,l:l,r:un()};}return l;}
function un(){if(pk()&&pk().t==='o'&&pk().v==='-'){nx();return {op:'neg',l:fc()};}return fc();}
function fc(){var t=nx();if(t.t==='n')return {n:t.v};if(t.t==='o'&&t.v==='('){var e=ex();nx();return e;}throw new Error(lang.name+': parse');}
return ex();}
function codegen(node,bc){bc=bc||[];if('n' in node){bc.push(['push',node.n]);return bc;}
if(node.op==='neg'){codegen(node.l,bc);bc.push(['neg']);return bc;}
codegen(node.l,bc);codegen(node.r,bc);bc.push([node.op]);return bc;}
function runBC(bc){var st=[];bc.forEach(function(ins){var op=ins[0];
if(op==='push')st.push(ins[1]);else if(op==='neg')st.push(-st.pop());
else{var b=st.pop(),a=st.pop();st.push(op==='+'?a+b:op==='-'?a-b:op==='*'?a*b:op==='/'?Math.floor(a/b):a%b);}});return st[0];}
function bodyOf(src){var m=src.match(lang.bodyRe);if(!m)throw new Error(lang.name+': no entry point');return m[1];}
return {compile:function(src){return codegen(parse(lex(bodyOf(src))));},run:function(src){return runBC(this.compile(src));},lang:lang.name};}
