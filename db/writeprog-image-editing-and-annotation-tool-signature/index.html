Working implementation

function imgNew(w,h,fill){var px=[];for(var y=0;y<h;y++){var row=[];for(var x=0;x<w;x++)row.push(fill||[0,0,0]);px.push(row);}return {w:w,h:h,px:px};}
function imgGet(im,x,y){return im.px[y][x];}
function imgSet(im,x,y,c){im.px[y][x]=c;return im;}
function imgInvert(im){im.px.forEach(function(row){row.forEach(function(p,i){row[i]=[255-p[0],255-p[1],255-p[2]];});});return im;}
function imgGrayscale(im){im.px.forEach(function(row){row.forEach(function(p,i){var g=Math.round(0.299*p[0]+0.587*p[1]+0.114*p[2]);row[i]=[g,g,g];});});return im;}
function imgAnnotateRect(im,x,y,w,h,color){for(var i=x;i<x+w;i++){imgSet(im,i,y,color);imgSet(im,i,y+h-1,color);}for(var j=y;j<y+h;j++){imgSet(im,x,j,color);imgSet(im,x+w-1,j,color);}return im;}
function imgAvg(im){var s=[0,0,0],n=im.w*im.h;im.px.forEach(function(row){row.forEach(function(p){s[0]+=p[0];s[1]+=p[1];s[2]+=p[2];});});return s.map(function(v){return Math.round(v/n);});}
