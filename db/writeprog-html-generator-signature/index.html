Working implementation

function el(tag,attrs,children){attrs=attrs||{};children=children||[];
var a=Object.keys(attrs).map(function(k){return ' '+k+'="'+String(attrs[k]).replace(/"/g,'&quot;')+'"';}).join('');
var inner=children.map(function(c){return typeof c==='string'?c:el(c.tag,c.attrs,c.children);}).join('');
var voidTags={img:1,br:1,hr:1,input:1,meta:1,link:1};
if(voidTags[tag])return '<'+tag+a+'>';return '<'+tag+a+'>'+inner+'</'+tag+'>';}
function page(title,bodyEls){return '<!DOCTYPE html>\n<html><head><meta charset="utf-8"><title>'+title+'</title></head><body>'+bodyEls.map(function(e){return el(e.tag,e.attrs,e.children);}).join('')+'</body></html>';}
