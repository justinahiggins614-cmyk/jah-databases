    """Single-pool Kt/V estimator (Daugirdas simplified). Target >= 1.2."""
    r = post_bun / pre_bun
    ktv = -__import__('math').log(r - 0.008 * hours) + (4 - 3.5 * r) * (uf_liters / weight_kg)
    return f"Kt/V {ktv:.2f} -> {'ADEQUATE (>=1.2)' if ktv >= 1.2 else 'INADEQUATE - review prescription'}"
print(ktv_adequacy(70, 22, 4, 2.5, 70))
