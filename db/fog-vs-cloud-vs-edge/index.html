// end-to-end: route three workloads to tiers by latency budget
const workloads=[{id:'train',budget_ms:600000},{id:'infer',budget_ms:15},{id:'filter',budget_ms:5}];
const tierOf=w=>w.budget_ms>1000?'cloud':(w.budget_ms>8?'fog':'edge');
const routed=workloads.map(w=>({id:w.id,tier:tierOf(w)}));
J({routed});assert(routed[0].tier==='cloud'&&routed[1].tier==='fog'&&routed[2].tier==='edge','routing');
