Working implementation

function ssNew(){return {cells:{}};}
function ssSet(sh,cell,val){sh.cells[cell.toUpperCase()]=val;return sh;}
function ssGet(sh,cell){var v=sh.cells[cell.toUpperCase()];if(typeof v==='string'&&v.charAt(0)==='='){return ssEval(sh,v.slice(1));}return (v===undefined||v===null)?0:v;}
function ssEval(sh,expr){var e=expr.replace(/[A-Za-z]+[0-9]+/g,function(c){return 'ssGet(sh,"'+c+'")';});return Function('ssGet','sh','return ('+e+')')(ssGet,sh);}
function ssCsv(sh){var cells=Object.keys(sh.cells);var maxR=0,maxC=0;cells.forEach(function(c){var m=c.match(/^([A-Z]+)([0-9]+)$/);if(m){maxC=Math.max(maxC,m[1].charCodeAt(0)-64);maxR=Math.max(maxR,parseInt(m[2],10));}});
var rows=[];for(var r=1;r<=maxR;r++){var row=[];for(var c=1;c<=maxC;c++){var cell=String.fromCharCode(64+c)+r;var v=ssGet(sh,cell);row.push(v===0&&!(cell in sh.cells)?'':String(v));}rows.push(row.join(','));}return rows.join('\n');}
