    items = {"identity confirmed": patient_ok, "site marked": site_marked,
             "antibiotics given": antibiotics, "counts correct": counts_ok,
             "specimen labeled": specimen_labeled}
    missing = [k for k, v in items.items() if not v]
    return ("PASS - proceed" if not missing else "HOLD - missing: " + ", ".join(missing))
print(surgery_checklist(True, True, True, True, False))
