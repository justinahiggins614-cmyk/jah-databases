import hashlib
def compute():
    created = 1922
    age = 2026 - created
    speakers = 500
    words = 20000
    phonemes = 24
    per_year = round(speakers / age, 1) if age else 0
    vocab_per_year = round(words / age, 2) if age else 0
    status = "living community language" if speakers >= 1000 else "small community or artistic language"
    return (f"created={created} age_years={age} speakers_est={speakers} "
            f"vocabulary={words} phonemes={phonemes} speakers_per_year={per_year} "
            f"vocab_per_year={vocab_per_year} status={status}")
def digest(text):
    return hashlib.sha256(text.encode()).hexdigest()
result = compute()
h1 = digest(result)
h2 = digest(compute())

print(result)
print("Run 1 SHA-256: " + h1)
