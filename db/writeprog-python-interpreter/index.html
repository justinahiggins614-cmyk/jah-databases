Working implementation

function pyLex(s){var toks=[],i=0;while(i<s.length){var c=s[i];if(/\s/.test(c)){i++;continue;}
if(/[0-9]/.test(c)){var j=i;while(j<s.length&&/[0-9.]/.test(s[j]))j++;toks.push({t:'num',v:parseFloat(s.slice(i,j))});i=j;continue;}
if('+-*/()%'.indexOf(c)>=0){toks.push({t:'op',v:c});i++;continue;}
if(/[A-Za-z_]/.test(c)){var j=i;while(j<s.length&&/[A-Za-z0-9_]/.test(s[j]))j++;toks.push({t:'id',v:s.slice(i,j)});i=j;continue;}
throw new Error('bad char '+c);}return toks;}
function pyParse(toks){var pos=0;
function peek(){return toks[pos];}function next(){return toks[pos++];}
function expr(){var l=term();while(peek()&&peek().t==='op'&&(peek().v==='+'||peek().v==='-')){var o=next().v;var r=term();l={op:o,l:l,r:r};}return l;}
function term(){var l=unary();while(peek()&&peek().t==='op'&&(peek().v==='*'||peek().v==='/'||peek().v==='%')){var o=next().v;var r=unary();l={op:o,l:l,r:r};}return l;}
function unary(){if(peek()&&peek().t==='op'&&peek().v==='-'){next();return {op:'neg',l:factor()};}return factor();}
function factor(){var t=next();if(t.t==='num')return {num:t.v};if(t.t==='op'&&t.v==='('){var e=expr();next();return e;}throw new Error('parse error');}
var e=expr();if(pos!==toks.length)throw new Error('trailing tokens');return e;}
function pyEval(node){if('num' in node)return node.num;if(node.op==='neg')return -pyEval(node.l);
var a=pyEval(node.l),b=pyEval(node.r);
if(node.op==='+')return a+b;if(node.op==='-')return a-b;if(node.op==='*')return a*b;if(node.op==='/')return a/b;if(node.op==='%')return a%b;throw new Error('op');}
function pyRun(src){return pyEval(pyParse(pyLex(src)));}
