Working implementation

var LEX_ES={hello:'hola',world:'mundo',good:'buenos',morning:'días',the:'el',cat:'gato',dog:'perro',is:'es',big:'grande',house:'casa',red:'rojo',water:'agua'};
function translateEnEs(text){return text.split(/(\s+)/).map(function(w){var k=w.toLowerCase().replace(/[.,!?]/,'');var t=LEX_ES[k];if(!t)return w;var p=w.match(/[.,!?]$/);return (/^[A-Z]/.test(w)?t.charAt(0).toUpperCase()+t.slice(1):t)+(p?p[0]:'');}).join('');}
function lexCoverage(text){var ws=text.toLowerCase().match(/[a-z]+/g)||[];var known=ws.filter(function(w){return LEX_ES[w];});return {total:ws.length,known:known.length};}
function lexAdd(en,es){LEX_ES[en.toLowerCase()]=es;}
