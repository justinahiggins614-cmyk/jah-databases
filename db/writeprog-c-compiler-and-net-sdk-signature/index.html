function parse(toks){var pos=0;function pk(){return toks[pos];}function nx(){return toks[pos++];}
function ex(){var l=tm();while(pk()&&pk().t==='o'&&(pk().v==='+'||pk().v==='-')){var o=nx().v;l={op:o,l:l,r:tm()};}return l;}

function codegen(node,bc){bc=bc||[];if('n' in node){bc.push(['push',node.n]);return bc;}
