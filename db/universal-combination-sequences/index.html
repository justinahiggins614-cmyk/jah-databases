import hashlib
import json

step_energies_j = [1.0, 2.0, 3.0]
budget_j = 10.0
def accumulate(energies):
    out, total = [], 0.0
    for e in energies:
        total += e
        out.append(total)
    return out
cumulative_a = accumulate(step_energies_j)
cumulative_b = accumulate(step_energies_j)
digest_a = hashlib.sha256(json.dumps(cumulative_a).encode("utf-8")).hexdigest()
digest_b = hashlib.sha256(json.dumps(cumulative_b).encode("utf-8")).hexdigest()
print("Sequence: " + 'The Universal Combination Sequence Field Guide Index')
for i, (e, c) in enumerate(zip(step_energies_j, cumulative_a), 1):
    print("  Step %d: +%r J  cumulative %r J" % (i, e, c))
print("Budget: %r J" % budget_j)
print("SHA-256 run 1: " + digest_a)
print("SHA-256 run 2: " + digest_b)
if digest_a == digest_b:
    print("Deterministic: MATCH - both runs produced the same balance.")
else:
    print("Deterministic: MISMATCH - the two runs disagreed.")
print('Sequence verdict: stable - every step stays within the budget.')
print('Note: this demo adds the stated step energies and compares them with the stated budget; it is arithmetic on the stated sequence, not a simulation of the real system.')
