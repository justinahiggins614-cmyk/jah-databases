Working demo program

import math
TITLE = 'White Sands Missile Range'
LAT, LON = 32.38, -106.48
REF_NAME, REF_LAT, REF_LON = 'Las Cruces, New Mexico', 32.31, -106.78
def haversine(lat1, lon1, lat2, lon2):
    r = math.radians
    dlat = r(lat2 - lat1); dlon = r(lon2 - lon1)
    a = math.sin(dlat/2)**2 + math.cos(r(lat1)) * math.cos(r(lat2)) * math.sin(dlon/2)**2
    return 6371.0 * 2 * math.asin(math.sqrt(a))
def to_dms(x, pos, neg):
    hemi = pos if x >= 0 else neg
    a = abs(x); d = int(a); m = int((a - d) * 60); s = round(((a - d) * 60 - m) * 60, 1)
    return '%d deg %d min %.1f sec %s' % (d, m, s, hemi)
def main():
    print(TITLE + ' - working demo')
    print('Approximate public coordinates: %.2f, %.2f' % (LAT, LON))
    print('Latitude DMS: ' + to_dms(LAT, 'N', 'S'))
    print('Longitude DMS: ' + to_dms(LON, 'E', 'W'))
    print('Distance to %s (%.2f, %.2f): %.1f km' % (REF_NAME, REF_LAT, REF_LON, haversine(LAT, LON, REF_LAT, REF_LON)))
