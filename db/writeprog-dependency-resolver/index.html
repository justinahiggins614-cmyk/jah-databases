Working implementation

function semParts(v){return v.split('.').map(Number);}
function semCmp(a,b){var A=semParts(a),B=semParts(b);for(var i=0;i<3;i++){if(A[i]!==B[i])return A[i]-B[i];}return 0;}
function semSat(v,range){var m=range.match(/^(>=|<=|>|<|=|\^|~)?(.+)$/);var op=m[1]||'=',ver=m[2];var c=semCmp(v,ver);
if(op==='=')return c===0;if(op==='>')return c>0;if(op==='>=')return c>=0;if(op==='<')return c<0;if(op==='<=')return c<=0;
if(op==='^'){var M=semParts(ver);return semCmp(v,ver)>=0&&semCmp(v,(M[0]+1)+'.0.0')<0;}
if(op==='~'){var M=semParts(ver);return semCmp(v,ver)>=0&&semCmp(v,M[0]+'.'+(M[1]+1)+'.0')<0;}return false;}
function depResolve(available,range){var cands=available.filter(function(v){return semSat(v,range);}).sort(semCmp);
return cands.length?cands[cands.length-1]:null;}
