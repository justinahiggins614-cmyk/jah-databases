Working implementation

function tfNew(){return {tests:[]};}
function tfAdd(tf,name,fn){tf.tests.push({name:name,fn:fn});return tf;}
function tfRun(tf){var passed=0,failed=[];tf.tests.forEach(function(t){try{t.fn();passed++;}catch(e){failed.push({name:t.name,error:String(e&&e.message||e)});}});
return {total:tf.tests.length,passed:passed,failed:failed,ok:failed.length===0};}
function tfAssert(cond,msg){if(!cond)throw new Error(msg||'assertion failed');}
function tfAssertEq(a,b,msg){if(a!==b)throw new Error((msg||'not equal')+': '+JSON.stringify(a)+' !== '+JSON.stringify(b));}
