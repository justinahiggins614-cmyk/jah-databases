Working implementation

function pfTime(fn,iters){iters=iters||100;var t0=process.hrtime.bigint();var last;
for(var i=0;i<iters;i++)last=fn(i);var t1=process.hrtime.bigint();
return {iters:iters,totalNs:Number(t1-t0),avgNs:+(Number(t1-t0)/iters).toFixed(1),last:last};}
function pfMem(fn){var b4=process.memoryUsage().heapUsed;var r=fn();var after=process.memoryUsage().heapUsed;
return {deltaBytes:after-b4,result:r};}
function pfCompare(a,b,iters){var ra=pfTime(a,iters),rb=pfTime(b,iters);
return {aAvgNs:ra.avgNs,bAvgNs:rb.avgNs,faster:ra.avgNs<=rb.avgNs?'a':'b',ratio:+(Math.max(ra.avgNs,rb.avgNs)/Math.max(1,Math.min(ra.avgNs,rb.avgNs))).toFixed(2)};}
