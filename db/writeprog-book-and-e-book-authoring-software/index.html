Working implementation

function bookNew(title,author){return {title:title,author:author,chapters:[]};}
function bookAddChapter(b,title,paras){b.chapters.push({title:title,paras:paras||[]});return b;}
function bookWords(b){return b.chapters.reduce(function(a,c){return a+c.paras.reduce(function(x,p){return x+p.split(/\s+/).filter(Boolean).length;},0);},0);}
function bookToc(b){return b.chapters.map(function(c,i){return (i+1)+'. '+c.title;});}
function bookRender(b){return '# '+b.title+'\n\n*by '+b.author+'*\n\n'+b.chapters.map(function(c,i){return '## Chapter '+(i+1)+': '+c.title+'\n\n'+c.paras.join('\n\n');}).join('\n\n');}
