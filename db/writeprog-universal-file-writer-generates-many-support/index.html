Working implementation

var FW_REGISTRY={};
function fwRegister(fmt,writer){FW_REGISTRY[fmt]=writer;}
function fwWrite(fmt,data){if(!FW_REGISTRY[fmt])throw new Error('unsupported format '+fmt);return FW_REGISTRY[fmt](data);}
function fwFormats(){return Object.keys(FW_REGISTRY);}
fwRegister('txt',function(d){return String(d.text||'');});
fwRegister('json',function(d){return JSON.stringify(d.value,null,2);});
fwRegister('csv',function(d){return d.rows.map(function(r){return r.join(',');}).join('\n');});
fwRegister('html',function(d){return '<!DOCTYPE html><html><body>'+d.body+'</body></html>';});
fwRegister('md',function(d){return '# '+d.title+'\n\n'+(d.body||'');});
