import math
def _gser(a, x):
    gln = math.lgamma(a); ap = a; s = 1.0 / a; d = s
    for _ in range(200):
        ap += 1; d *= x / ap; s += d
        if abs(d) < abs(s) * 1e-12: break
    return s * math.exp(-x + a * math.log(x) - gln)
def _gcf(a, x):
    gln = math.lgamma(a); b = x + 1 - a; c = 1e30; d = 1 / b; h = d
    for i in range(1, 200):
        an = -i * (i - a); b += 2; d = an * d + b
        if abs(d) < 1e-30: d = 1e-30
        c = b + an / c
        if abs(c) < 1e-30: c = 1e-30
        d = 1 / d; de = d * c; h *= de

    gln = math.lgamma(a); ap = a; s = 1.0 / a; d = s
    for _ in range(200):
        ap += 1; d *= x / ap; s += d
        if abs(d) < abs(s) * 1e-12: break
    return s * math.exp(-x + a * math.log(x) - gln)
def _gcf(a, x):
    gln = math.lgamma(a); b = x + 1 - a; c = 1e30; d = 1 / b; h = d
    for i in range(1, 200):
        an = -i * (i - a); b += 2; d = an * d + b
        if abs(d) < 1e-30: d = 1e-30
        c = b + an / c
        if abs(c) < 1e-30: c = 1e-30
        d = 1 / d; de = d * c; h *= de
        if abs(de - 1) < 1e-12: break
    return math.exp(-x + a * math.log(x) - gln) * h
def chi2_p(x, k):
    a = k / 2.0; xx = x / 2.0
    if xx == 0: return 1.0
    if xx < a + 1: return 1 - _gser(a, xx)
    return _gcf(a, xx)
