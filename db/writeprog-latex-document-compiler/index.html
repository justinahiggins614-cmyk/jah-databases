Working implementation

function latexToHtml(s){var h=s;
h=h.replace(/\\section\{([^}]*)\}/g,'<h2>$1</h2>').replace(/\\subsection\{([^}]*)\}/g,'<h3>$1</h3>');
h=h.replace(/\\textbf\{([^}]*)\}/g,'<strong>$1</strong>').replace(/\\textit\{([^}]*)\}/g,'<em>$1</em>').replace(/\\underline\{([^}]*)\}/g,'<u>$1</u>');
h=h.replace(/\\begin\{itemize\}([\s\S]*?)\\end\{itemize\}/g,function(m,b){return '<ul>'+b.replace(/\\item\s*/g,'<li>')+'</ul>';});
h=h.replace(/\$(.+?)\$/g,'<span class="math">$1</span>');
h=h.replace(/\\\\/g,'<br>');return h;}
function latexStrip(s){return s.replace(/\\[a-zA-Z]+\{([^}]*)\}/g,'$1').replace(/\\[a-zA-Z]+/g,'').replace(/[{}]/g,'');}
