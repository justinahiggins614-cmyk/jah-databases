design_title = 'The Internet Courtroom Field Guide Index'
design = 'A play courtroom website: a visitor picks a case, such as the missing biscuit or the borrowed umbrella, and the courtroom hears it in plain words, with a claim, a defence, and a playful verdict, all labeled as play. Its verdicts are not legally recognized, it does not override any real court, and no real dispute is ever decided here. This record is the field guide index for the whole database: it lists all twenty-three content records in ladder order, simple to compounded to complex, and tells a reader where to start.'
steps = ['Add a Signature tournament bracket with rounds, divisions, and a season', 'Add a Signature creator studio so visitors can build a new play trial', 'Add a Signature review step so every new play trial is checked before it is published', 'Add the Signature play trial in many languages, with the same rules in each', 'Add a Signature offline pack so the play trial works with no connection']

print("Website design study: " + design_title)

    print("Step %d: %s" % (i, step))
