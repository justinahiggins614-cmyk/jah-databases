Working implementation

var NL_RULES=[
{re:/add (\w+) and (\w+)/i, code:function(m){return m[1]+' + '+m[2];}},
{re:/multiply (\w+) by (\w+)/i, code:function(m){return m[1]+' * '+m[2];}},
{re:/print (.+)/i, code:function(m){return 'console.log('+m[1]+');';}},
{re:/set (\w+) to (.+)/i, code:function(m){return 'let '+m[1]+' = '+m[2]+';';}},
{re:/if (\w+) is greater than (\w+)/i, code:function(m){return 'if ('+m[1]+' > '+m[2]+') {';}}];
function nlToCode(text){for(var i=0;i<NL_RULES.length;i++){var m=text.match(NL_RULES[i].re);if(m)return {ok:true,code:NL_RULES[i].code(m),rule:i};}return {ok:false,code:null};}
function nlBatch(lines){return lines.map(nlToCode);}
