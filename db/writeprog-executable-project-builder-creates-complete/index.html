Working implementation

function xbBuild(spec){var files={};
files['src/main.js']="'use strict';\n// "+spec.name+"\nfunction main(){\n  console.log('"+spec.name+" v"+spec.version+"');\n}\nmain();\n";
files['build.sh']='#!/bin/sh\nset -e\nnode src/main.js\n';
files['README.md']='# '+spec.name+'\n\nRun: `sh build.sh`\n';
files['.gitignore']='node_modules/\n';
return {name:spec.name,files:files};}
function xbValidate(proj){var errs=[];['src/main.js','build.sh','README.md'].forEach(function(f){if(!(f in proj.files))errs.push('missing '+f);});
try{new (require('vm').Script)(proj.files['src/main.js']);}catch(e){errs.push('main.js invalid: '+e.message);}
return {ok:errs.length===0,errors:errs};}
function xbRunMain(proj){return require('vm').runInNewContext(proj.files['src/main.js']+'; "ran";');}
