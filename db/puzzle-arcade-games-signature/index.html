Gemfall Cascade Rank 1 · simple · puzzle · beginner

Gemfall Cascade is an original puzzle game. Colored gems rain into a valley well, and players stack them into matching rows before the well fills to the brim.

Pieces arrive one at a time: slide left or right to choose a column, rotate to change the shape, and drop to place it. Filling a row across the board clears it and scores points, while an uneven board grows harder to manage with every piece.

Rotating a piece before sliding it usually beats sliding first, since the rotated shape often fits a gap the flat shape cannot.

Original game note: This is an original arcade-style game created for this Signature edition, with its own title, rules, and play.

How to play

Type left, right, rotate or drop for each piece, or quit to stop.

Working demo program

Save the code below as game.py and run python3 game.py --play to play it. Running it without arguments replays the scripted session whose output is stored with the record.

import sys
PARAMS = {'pieces': ['red drop', 'blue drop', 'green drop', 'yellow drop', 'red drop', 'blue drop', 'green drop', 'red drop']}
AUTO = [['rotate', 'drop'], ['right', 'drop'], ['right', 'right', 'drop'], ['right', 'right', 'right', 'drop'], ['left', 'left', 'rotate', 'drop'], ['left', 'drop'], ['drop'], ['right', 'drop']]

def auto_play():
    heights = [0] * 6
    score, lines = 0, 0
    print("Puzzle tribute: auto session")
    for pi, piece in enumerate(PARAMS["pieces"]):
        cmds = AUTO[pi] if pi < len(AUTO) else ["drop"]
        col, vertical = 2, False
        for cmd in cmds:
            if cmd == "left":
                col = max(0, col - 1)
            elif cmd == "right":
                col = min(5, col + 1)
            elif cmd == "rotate":
                vertical = not vertical
            elif cmd == "drop":
                break
        if vertical:
            heights[col] += 2
        else:
            if col >= 5:
                col = 4
            heights[col] += 1
            heights[col + 1] += 1
        cleared = 0
        while min(heights) > 0:
            heights = [h - 1 for h in heights]
            cleared += 1
        lines += cleared
        score += cleared * 100 + 10
        print("Piece %d: %s col %d %s heights %s score %d" % (pi + 1, piece, col, "vertical" if vertical else "flat", heights, score))
    print("FINAL SCORE: %d" % score)
    print("RESULT: LINES %d" % lines)

def play():
    heights = [0] * 6
    score, lines = 0, 0
    print("Puzzle tribute: type left/right/rotate/drop per piece, or quit. Fill rows to clear them.")
    pi = 0
    col, vertical = 2, False
    for line in sys.stdin:
        cmd = line.strip().lower()
        if cmd == "quit":
            break
        if pi >= len(PARAMS["pieces"]):
            break
        if cmd == "left":
            col = max(0, col - 1)
        elif cmd == "right":
            col = min(5, col + 1)
        elif cmd == "rotate":
            vertical = not vertical
        elif cmd == "drop":
            piece = PARAMS["pieces"][pi]
            if vertical:
                heights[col] += 2
            else:
                if col >= 5:
                    col = 4
                heights[col] += 1
                heights[col + 1] += 1
            cleared = 0
            while min(heights) > 0:
                heights = [h - 1 for h in heights]
                cleared += 1
            lines += cleared
            score += cleared * 100 + 10
            print("Piece %d: %s heights %s score %d" % (pi + 1, piece, heights, score))
            pi += 1
            col, vertical = 2, False
        else:
            print("Unknown command: %s" % cmd)
    print("FINAL SCORE: %d" % score)
    print("RESULT: LINES %d" % lines)

if __name__ == "__main__":
    if "--play" in sys.argv:
        play()
    else:
        auto_play()
