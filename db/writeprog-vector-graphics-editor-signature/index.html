function vgPath(d){return {d:d};}
function vgTranslate(p,dx,dy){return vgPath(p.d.replace(/([ML])\s*(-?[\d.]+)[,\s]+(-?[\d.]+)/g,function(m,c,x,y){return c+' '+(+x+dx)+' '+(+y+dy);}));}
function vgScale(p,sx,sy){sy=sy===undefined?sx:sy;return vgPath(p.d.replace(/([ML])\s*(-?[\d.]+)[,\s]+(-?[\d.]+)/g,function(m,c,x,y){return c+' '+(+x*sx)+' '+(+y*sy);}));}
function vgBounds(p){var nums=p.d.match(/-?[\d.]+/g).map(Number);var xs=[],ys=[];for(var i=0;i<nums.length;i+=2){xs.push(nums[i]);ys.push(nums[i+1]);}
return {minX:Math.min.apply(null,xs),maxX:Math.max.apply(null,xs),minY:Math.min.apply(null,ys),maxY:Math.max.apply(null,ys)};}
function vgSvg(p,w,h){return '<svg xmlns="http://www.w3.org/2000/svg" width="'+w+'" height="'+h+'"><path d="'+p.d+'" fill="none" stroke="black"/></svg>';}
