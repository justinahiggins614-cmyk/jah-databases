# The JAH universal finder: one machine, three probes, three targets
import math
def respond(im, probe):
    h,w=len(im),len(im[0]); m=[[0.0]*w for _ in range(h)]
    for y in range(1,h-1):
        for x in range(1,w-1):
            m[y][x]=sum(im[y+dy-1][x+dx-1]*probe[dy][dx] for dy in range(3) for dx in range(3))
    return m
def finder(image, probe, thresh, name):
    m=respond(image, probe)
    peak=max(max(row) for row in m)
    yx=max(((m[y][x],(y,x)) for y in range(len(m)) for x in range(len(m[0]))))[1]
    return "%-12s peak=%6.1f at %s -> %s" % (name, peak, yx, "FOUND" if peak>thresh else "absent")
EDGE=[[-1,0,1],[-2,0,2],[-1,0,1]]
BLOB=[[0,1,0],[1,-4,1],[0,1,0]]
LINE=[[-1,-1,-1],[2,2,2],[-1,-1,-1]]
cat=[[0]*9 for _ in range(9)]
for y in range(9):
    for x in range(9):
        if (x-4)**2+(y-5)**2<=9: cat[y][x]=10
dots=[[0]*9 for _ in range(9)]; dots[4][4]=50
bar=[[0]*9 for _ in range(9)]
for x in range(9): bar[4][x]=30
print(finder(cat, EDGE, 20, "edges->cat"))
print(finder(dots, BLOB, 20, "blob->dot"))
print(finder(bar, LINE, 20, "line->bar"))
print("Signature framework: probe, map, decide — the same machine for every target.")
