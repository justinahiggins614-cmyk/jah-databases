Starfall Sentinels Rank 1 · simple · fixed shooter · beginner

A lone orbital lighthouse keeper defends a chain of signal buoys from drifting ember swarms that descend in orderly ranks. Each buoy saved keeps a shipping lane lit, and the keeper's lamp-cannon is the only weapon standing between the dark and the lanes.

The rules are simple: the player slides a cannon along the bottom of a 7-column grid, fires upward at 3 rows of descending foes, and scores for every foe cleared before the formation reaches row 11. Clearing a full column without losing ground earns a bonus, and the run ends when the formation lands or is destroyed.

The lamp-cannon fires straight up and the ember ranks march downward, so the keeper wins by sweeping columns methodically and saving a retreat lane under the last buoy. Scoring rewards full columns cleared without losing a buoy, and a perfect sweep lights a bonus beacon.

PARAMS = {'cols': 7, 'rows': 3, 'start_col': 1, 'floor_row': 11}
AUTO = ['left', 'shoot', 'shoot', 'shoot', 'right', 'shoot', 'shoot', 'shoot', 'right', 'shoot', 'shoot', 'shoot', 'right', 'shoot', 'shoot', 'shoot', 'right', 'shoot', 'shoot', 'shoot', 'right', 'shoot', 'shoot', 'shoot', 'right', 'shoot', 'shoot', 'shoot']

def auto_play():
    cols = PARAMS["cols"]
    invaders = set()
    for r in range(PARAMS["rows"]):
        for c in range(cols):
            invaders.add((r, c))
    player = PARAMS["start_col"]
    score = 0
    turn = 0
    print("Fixed shooter tribute: auto session")
    for cmd in AUTO:
        turn += 1
        if cmd == "left":
            player = max(0, player - 1)
        elif cmd == "right":
            player = min(cols - 1, player + 1)
        elif cmd == "shoot":
            col_hits = [(r, c) for (r, c) in invaders if c == player]
            if col_hits:
                invaders.discard(max(col_hits))
                score += 50
        if turn % 3 == 0 and invaders:
            invaders = set((r + 1, c) for (r, c) in invaders)
        print("Turn %d: %s player col %d invaders %d score %d" % (turn, cmd, player, len(invaders), score))
        if not invaders or any(r >= PARAMS["floor_row"] for (r, c) in invaders):
            break
    deepest = max([r for (r, c) in invaders]) if invaders else -1
    print("DEEPEST ROW: %d / FLOOR: %d" % (deepest, PARAMS["floor_row"]))
    print("FINAL SCORE: %d" % score)
    if not invaders:
        print("RESULT: WIN")
    elif deepest >= PARAMS["floor_row"]:

def auto_play():
    cols = PARAMS["cols"]
    invaders = set()
    for r in range(PARAMS["rows"]):
        for c in range(cols):
            invaders.add((r, c))
    player = PARAMS["start_col"]
    score = 0
    turn = 0
    print("Fixed shooter tribute: auto session")
    for cmd in AUTO:
        turn += 1
        if cmd == "left":
            player = max(0, player - 1)
        elif cmd == "right":
            player = min(cols - 1, player + 1)
        elif cmd == "shoot":
            col_hits = [(r, c) for (r, c) in invaders if c == player]
            if col_hits:
                invaders.discard(max(col_hits))
                score += 50
        if turn % 3 == 0 and invaders:
            invaders = set((r + 1, c) for (r, c) in invaders)
        print("Turn %d: %s player col %d invaders %d score %d" % (turn, cmd, player, len(invaders), score))
        if not invaders or any(r >= PARAMS["floor_row"] for (r, c) in invaders):
            break
    deepest = max([r for (r, c) in invaders]) if invaders else -1
    print("DEEPEST ROW: %d / FLOOR: %d" % (deepest, PARAMS["floor_row"]))
    print("FINAL SCORE: %d" % score)
    if not invaders:
        print("RESULT: WIN")
    elif deepest >= PARAMS["floor_row"]:
        print("RESULT: OVERRUN")
    else:
        print("RESULT: SESSION END")
