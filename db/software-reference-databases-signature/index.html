The Python Package Index, PyPI, is the place Python goes to find its own software. PyPI is operated by the Python Software Foundation, and its software, called Warehouse, is open source under the Apache License 2.0, while the packages themselves carry whatever license their authors chose, most of them open source. import hashlib
import json
import random

title = 'The Apple App Store Catalog'
points = [('Apps in the catalog, approx.', 1800000), ('Apps at the opening in July 2008', 500), ('Year the App Store opened', 2008), ('Developer program fee, dollars per year', 99)]
seed = 2008
claimed_note = 'As reported: the App Store opened in July 2008 with 500 apps and holds more than 1.8 million apps today.'

def compute():
    rng = random.Random(seed)
    sample = rng.choice([p[0] for p in points])
    values = [p[1] for p in points]
    total = sum(values)
    mean = total / len(values)
    lo = min(values)
    hi = max(values)
    ratio = hi / lo if lo else 0
    return {"title": title, "points": len(points), "total": total,
            "mean": round(mean, 3), "min": lo, "max": hi,
            "max_min_ratio": round(ratio, 3), "spot_check": sample}

result_a = compute()
result_b = compute()
digest_a = hashlib.sha256(json.dumps(result_a, sort_keys=True).encode("utf-8")).hexdigest()
digest_b = hashlib.sha256(json.dumps(result_b, sort_keys=True).encode("utf-8")).hexdigest()
print("Record figures check: " + title)
for label, value in points:

import json
import random

title = 'The Conda Package Index'

def compute():
    rng = random.Random(seed)
    sample = rng.choice([p[0] for p in points])
    values = [p[1] for p in points]

result_b = compute()
