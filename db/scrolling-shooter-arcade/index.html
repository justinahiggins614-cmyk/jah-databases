Xevious Rank 1 · simple · vertical scrolling shooter · beginner

Xevious, Namco's 1982 vertical scrolling shooter, set the template for the genre: the player flies a Solvalou fighter over scrolling terrain, firing at air targets while bombing ground targets with a separate sight. The terrain scrolls continuously beneath the ship, and the game's mix of air and ground combat, plus its hidden towers and flags, gave it unusual depth.

The tribute demo flies a scrolling course starting on row 2, with enemies spawning on ticks 2, 6, 10, 14: the player moves between rows and shoots, surviving the scroll while picking off enemies before they reach the ship. It is an original miniature inspired by the scrolling-shooter idea, not a copy of the arcade game.

The bomb sight leads the ship, so ground targets must be bombed before they pass under the fighter, not as they arrive. Air targets are best met with short bursts, since the fighter's fire is narrow and the sky fills quickly.

Playable tribute note: The demo below is an original tribute inspired by the mechanics of Xevious (1982). It is not Xevious (1982), and it uses no art, music, or code from that game.

How to play

Type up/down/left/right/shoot each tick to survive the scroll, quit to stop.

import sys
PARAMS = {'start_row': 2, 'spawns': [[2, 1], [6, 3], [10, 2], [14, 4]]}
AUTO = ['shoot', 'shoot', 'up', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot']

def auto_play():
    player = [2, PARAMS["start_row"]]
    enemies = []
    schedule = dict((t, r) for t, r in PARAMS["spawns"])
    score, lives = 0, 3
    print("Scrolling tribute: auto session")
    for tick, cmd in enumerate(AUTO):
        if tick in schedule:
            enemies.append([11, schedule[tick]])
        if cmd == "up":
            player[1] = max(0, player[1] - 1)
        elif cmd == "down":
            player[1] = min(5, player[1] + 1)
        elif cmd == "left":
            player[0] = max(0, player[0] - 1)
        elif cmd == "right":
            player[0] = min(11, player[0] + 1)
        elif cmd == "shoot":
            for e in list(enemies):
                if e[1] == player[1] and player[0] < e[0] <= player[0] + 4:
                    enemies.remove(e)
                    score += 100
                    break
        enemies = [[x - 1, y] for (x, y) in enemies]
        if any((x, y) == (player[0], player[1]) for (x, y) in enemies):
            lives -= 1
            enemies = [e for e in enemies if (e[0], e[1]) != (player[0], player[1])]
        score += 10
        print("Tick %d: %s player %s enemies %d score %d lives %d" % (tick + 1, cmd, tuple(player), len(enemies), score, lives))
        if lives <= 0:
            break
    print("FINAL SCORE: %d" % score)
    print("RESULT: %s" % ("SURVIVED" if lives > 0 else "OUT OF LIVES"))

AUTO = ['shoot', 'shoot', 'shoot', 'up', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot', 'shoot']

def auto_play():
    player = [2, PARAMS["start_row"]]
    enemies = []
    schedule = dict((t, r) for t, r in PARAMS["spawns"])
    score, lives = 0, 3
    print("Scrolling tribute: auto session")
    for tick, cmd in enumerate(AUTO):
        if tick in schedule:
            enemies.append([11, schedule[tick]])
        if cmd == "up":
            player[1] = max(0, player[1] - 1)
        elif cmd == "down":
            player[1] = min(5, player[1] + 1)
        elif cmd == "left":
            player[0] = max(0, player[0] - 1)
        elif cmd == "right":
            player[0] = min(11, player[0] + 1)
        elif cmd == "shoot":
            for e in list(enemies):
                if e[1] == player[1] and player[0] < e[0] <= player[0] + 4:
                    enemies.remove(e)
                    score += 100
                    break
        enemies = [[x - 1, y] for (x, y) in enemies]
        if any((x, y) == (player[0], player[1]) for (x, y) in enemies):
            lives -= 1
            enemies = [e for e in enemies if (e[0], e[1]) != (player[0], player[1])]
        score += 10
        print("Tick %d: %s player %s enemies %d score %d lives %d" % (tick + 1, cmd, tuple(player), len(enemies), score, lives))
        if lives <= 0:
            break
    print("FINAL SCORE: %d" % score)
    print("RESULT: %s" % ("SURVIVED" if lives > 0 else "OUT OF LIVES"))

def play():
    player = [2, PARAMS["start_row"]]
    enemies = []
    schedule = dict((t, r) for t, r in PARAMS["spawns"])
    score, lives, tick = 0, 3, 0
    print("Scrolling tribute: type up/down/left/right/shoot each tick, or quit. Survive the scroll.")
    for line in sys.stdin:
        cmd = line.strip().lower()
        if cmd == "quit":
            break
        if cmd not in ("up", "down", "left", "right", "shoot"):
            print("Unknown command: %s" % cmd)
            continue
        if tick in schedule:
            enemies.append([11, schedule[tick]])
        if cmd == "up":
            player[1] = max(0, player[1] - 1)
        elif cmd == "down":
            player[1] = min(5, player[1] + 1)
        elif cmd == "left":
            player[0] = max(0, player[0] - 1)
        elif cmd == "right":
            player[0] = min(11, player[0] + 1)
        elif cmd == "shoot":
            for e in list(enemies):
                if e[1] == player[1] and player[0] < e[0] <= player[0] + 4:
                    enemies.remove(e)
                    score += 100
                    break
        enemies = [[x - 1, y] for (x, y) in enemies]
        if any((x, y) == (player[0], player[1]) for (x, y) in enemies):
            lives -= 1
            enemies = [e for e in enemies if (e[0], e[1]) != (player[0], player[1])]
        score += 10
        tick += 1
        print("player %s enemies %d score %d lives %d" % (tuple(player), len(enemies), score, lives))
        if lives <= 0:
            break
    print("FINAL SCORE: %d" % score)
    print("RESULT: %s" % ("SURVIVED" if lives > 0 else "OUT OF LIVES" if tick else "QUIT"))

if __name__ == "__main__":
    if "--play" in sys.argv:
        play()
    else:
        auto_play()
