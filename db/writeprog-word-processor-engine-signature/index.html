Working implementation

function wpDoc(){return {paras:[]};}
function wpAdd(doc,text,style){doc.paras.push({text:text,style:style||'normal'});return doc;}
function wpWords(doc){return doc.paras.reduce(function(a,p){return a+p.text.split(/\s+/).filter(function(w){return w;}).length;},0);}
function wpExport(doc){return doc.paras.map(function(p){return p.text;}).join('\n\n');}
function wpFindReplace(doc,find,repl){doc.paras.forEach(function(p){p.text=p.text.split(find).join(repl);});return doc;}
