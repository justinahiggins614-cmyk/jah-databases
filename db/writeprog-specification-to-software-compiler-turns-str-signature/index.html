function sscCompile(spec){var fns=spec.functions.map(function(f){var params=(f.params||[]).join(', ');
var body=f.returns||'null';return 'function '+f.name+'('+params+'){ return ('+body+'); }';}).join('\n');
var exports='module.exports = { '+spec.functions.map(function(f){return f.name;}).join(', ')+' };';
return "'use strict';\n// generated from spec: "+spec.name+'\n'+fns+'\n'+exports;}
function sscLoad(code){var m={exports:{}};new Function('module','exports',code)(m,m.exports);return m.exports;}
function sscTest(lib,spec){return spec.functions.map(function(f){var got=lib[f.name].apply(null,f.testArgs||[]);return {name:f.name,pass:JSON.stringify(got)===JSON.stringify(f.expected),got:got};});}
