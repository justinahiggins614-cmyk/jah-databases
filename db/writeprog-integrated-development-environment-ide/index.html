Working implementation

function ideProject(name){return {name:name,files:{},tasks:{}};}
function ideAddFile(p,name,content){p.files[name]=content;return p;}
function ideAddTask(p,name,fn){p.tasks[name]=fn;return p;}
function ideRunTask(p,name){if(!(name in p.tasks))return {ok:false,error:'no such task'};try{return {ok:true,result:p.tasks[name](p)};}catch(e){return {ok:false,error:String(e&&e.message||e)};}}
function ideFileList(p){return Object.keys(p.files);}
function ideWordCount(p){var n=0;Object.keys(p.files).forEach(function(f){n+=p.files[f].split(/\s+/).filter(Boolean).length;});return n;}
