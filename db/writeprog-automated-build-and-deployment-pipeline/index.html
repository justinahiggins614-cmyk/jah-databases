Working implementation

function pipeNew(){return {stages:[]};}
function pipeStage(p,name,fn){p.stages.push({name:name,fn:fn});return p;}
function pipeRun(p,ctx){var log=[];for(var i=0;i<p.stages.length;i++){var s=p.stages[i];log.push('start '+s.name);
try{var r=s.fn(ctx);log.push('ok '+s.name);ctx[s.name]=r;}catch(e){log.push('FAIL '+s.name+': '+e.message);return {ok:false,stage:s.name,log:log,ctx:ctx};}}
return {ok:true,log:log,ctx:ctx};}
function pipeSummary(res){return {ok:res.ok,stages:res.log.filter(function(l){return l.indexOf('ok ')>-1;}).length,failedStage:res.stage||null};}
