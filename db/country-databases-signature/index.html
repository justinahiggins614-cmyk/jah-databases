Working demo program

import hashlib
import json
import math
import random

title = 'The World Bank World Development Indicators'
entries = 1400
categories = ['Poverty', 'Health', 'Education', 'Energy', 'Trade']
trials = 25
seed = 71
claimed_note = 'Published holdings: more than 1,400 indicators for more than 200 economies; the demo samples 25 entries by indicator group and does not query the live database.'

def run_once():
    rng = random.Random(seed)
    counts = {c: 0 for c in categories}
    for _ in range(trials):
        counts[rng.choice(categories)] += 1
    return counts

expected = trials / len(categories)
counts_a = run_once()
counts_b = run_once()
chi2 = sum((o - expected) ** 2 / expected for o in counts_a.values())
result = {"title": title, "entries": entries, "trials": trials,
          "categories": len(categories), "seed": seed,
          "counts": counts_a, "chi_square": round(chi2, 3)}
digest_a = hashlib.sha256(json.dumps(result, sort_keys=True).encode("utf-8")).hexdigest()
digest_b = hashlib.sha256(json.dumps(result, sort_keys=True).encode("utf-8")).hexdigest()
print("Catalog: " + title)
print("Entries (published, approximate): %d" % entries)
print("Spot-check sample: %d entries across %d categories (fixed seed %d)" % (trials, len(categories), seed))
print("Category counts: " + json.dumps(counts_a, sort_keys=True))
print("Expected per category: %.2f entries" % expected)
print("Chi-square: %.3f (sum of (observed minus expected) squared, divided by expected)" % chi2)
print("Spot-check note: " + claimed_note)
print("SHA-256 run 1: " + digest_a)
print("SHA-256 run 2: " + digest_b)
if digest_a == digest_b:
    print("Deterministic: MATCH - both runs produced the same sampled counts.")
else:
    print("Deterministic: MISMATCH - the two runs disagreed.")
print("Note: this demo simulates a seeded spot-check sample only. It does not query the live database, and a balanced sample is chance, not a verification of the live holdings.")
print("Design study: this record describes a Signature database design, not an established public record.")

import hashlib
import json
import math
import random

title = 'The United Nations Data Portal'
entries = 60
categories = ['Population', 'Trade', 'Health', 'Education', 'Energy']
trials = 25
seed = 73
claimed_note = 'Published holdings: more than 60 databases from UN agencies; the demo samples 25 entries by topic and does not query the live database.'

def run_once():
    rng = random.Random(seed)
    counts = {c: 0 for c in categories}
    for _ in range(trials):
        counts[rng.choice(categories)] += 1
    return counts

expected = trials / len(categories)
counts_a = run_once()
counts_b = run_once()
chi2 = sum((o - expected) ** 2 / expected for o in counts_a.values())
result = {"title": title, "entries": entries, "trials": trials,
          "categories": len(categories), "seed": seed,

import json
import math
import random

entries = 200
categories = ['Exports', 'Imports', 'Products', 'Partners', 'Values']
trials = 25
seed = 81

def run_once():

    counts = {c: 0 for c in categories}

    return counts

counts_b = run_once()

Working demo program

import hashlib
import json
import math
import random

title = 'The World Bank Doing Business Archive'
entries = 190
categories = ['Starting a business', 'Permits', 'Credit', 'Contracts', 'Rankings']
trials = 25
seed = 86
claimed_note = 'Published holdings: annual rankings for 190 economies, 2003 to 2020, archived after discontinuation; the demo samples 25 entries by indicator group and does not query the live database.'

def run_once():
    rng = random.Random(seed)
    counts = {c: 0 for c in categories}
    for _ in range(trials):
        counts[rng.choice(categories)] += 1
    return counts

expected = trials / len(categories)
counts_a = run_once()
counts_b = run_once()
chi2 = sum((o - expected) ** 2 / expected for o in counts_a.values())
result = {"title": title, "entries": entries, "trials": trials,
          "categories": len(categories), "seed": seed,
          "counts": counts_a, "chi_square": round(chi2, 3)}
digest_a = hashlib.sha256(json.dumps(result, sort_keys=True).encode("utf-8")).hexdigest()
digest_b = hashlib.sha256(json.dumps(result, sort_keys=True).encode("utf-8")).hexdigest()
print("Catalog: " + title)
print("Entries (published, approximate): %d" % entries)
print("Spot-check sample: %d entries across %d categories (fixed seed %d)" % (trials, len(categories), seed))
print("Category counts: " + json.dumps(counts_a, sort_keys=True))
print("Expected per category: %.2f entries" % expected)
print("Chi-square: %.3f (sum of (observed minus expected) squared, divided by expected)" % chi2)
print("Spot-check note: " + claimed_note)
print("SHA-256 run 1: " + digest_a)
print("SHA-256 run 2: " + digest_b)
if digest_a == digest_b:
    print("Deterministic: MATCH - both runs produced the same sampled counts.")
else:
    print("Deterministic: MISMATCH - the two runs disagreed.")
print("Note: this demo simulates a seeded spot-check sample only. It does not query the live database, and a balanced sample is chance, not a verification of the live holdings.")
print("Design study: this record describes a Signature database design, not an established public record.")

import hashlib
import json
import math
import random

title = 'The ISO 3166 Country Code Registry'
entries = 249
categories = ['Two-letter codes', 'Three-letter codes', 'Numeric codes', 'Retired codes', 'Names']
trials = 25
seed = 91
claimed_note = 'Published holdings: 249 country and territory codes in three forms; the demo samples 25 entries by code form and does not query the live database.'

def run_once():
    rng = random.Random(seed)
    counts = {c: 0 for c in categories}
    for _ in range(trials):
        counts[rng.choice(categories)] += 1
    return counts

expected = trials / len(categories)
counts_a = run_once()
counts_b = run_once()
chi2 = sum((o - expected) ** 2 / expected for o in counts_a.values())
result = {"title": title, "entries": entries, "trials": trials,
          "categories": len(categories), "seed": seed,
          "counts": counts_a, "chi_square": round(chi2, 3)}
digest_a = hashlib.sha256(json.dumps(result, sort_keys=True).encode("utf-8")).hexdigest()
digest_b = hashlib.sha256(json.dumps(result, sort_keys=True).encode("utf-8")).hexdigest()
