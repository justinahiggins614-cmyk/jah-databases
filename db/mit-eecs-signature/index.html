xs = [1, 3, 5, 7, 9, 11, 13]
target, lo, hi, steps = 11, 0, len(xs) - 1, 0

    mid = (lo + hi) // 2
    steps += 1
    if xs[mid] == target:
        break
    elif xs[mid] < target:
        lo = mid + 1
    else:
        hi = mid - 1
