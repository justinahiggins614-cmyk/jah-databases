Working implementation

function saAnalyze(code){var issues=[];var lines=code.split('\n');
lines.forEach(function(l,i){var n=i+1;
if(/(^|[^A-Za-z_$])eval\s*\(/.test(l))issues.push({line:n,sev:'high',rule:'no-eval',msg:'eval() is dangerous'});
if(/child_process/.test(l))issues.push({line:n,sev:'high',rule:'child-process',msg:'spawning processes needs review'});
if(/(password|secret|api[_-]?key)\s*=\s*['"][^'"]+['"]/i.test(l))issues.push({line:n,sev:'high',rule:'hardcoded-secret',msg:'possible hardcoded secret'});
if(/[^=!]==[^=]/.test(l))issues.push({line:n,sev:'low',rule:'eqeqeq',msg:'use ==='});
var f=l.match(/function\s+\w+\s*\([^)]*\)\s*\{/);if(f)issues.push({line:n,sev:'info',rule:'fn-found',msg:'function defined'});});
return issues;}
function saSummary(issues){var s={high:0,low:0,info:0};issues.forEach(function(x){s[x.sev]=(s[x.sev]||0)+1;});return s;}
