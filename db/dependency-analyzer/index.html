def has_cycle():
    done = set()
    def visit(p, stack):
        if p in stack:
            return True
        if p in done:
            return False
        stack = stack + (p,)
        for n in parts.get(p, []):
            if visit(n, stack):
                return True
        done.add(p)
        return False
