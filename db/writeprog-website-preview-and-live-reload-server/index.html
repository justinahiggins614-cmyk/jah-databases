Working implementation

var http=require('http');
function previewServer(files){var server=http.createServer(function(req,res){var p=req.url==='/'?'/index.html':req.url;
if(files[p]){var ct=p.slice(-5)==='.html'?'text/html':p.slice(-4)==='.css'?'text/css':'text/plain';res.writeHead(200,{'Content-Type':ct});res.end(files[p]);}
else{res.writeHead(404);res.end('not found');}});return server;}
function httpGet(port,path){return new Promise(function(res,rej){http.get({port:port,path:path},function(r){var b='';r.on('data',function(c){b+=c;});r.on('end',function(){res({status:r.statusCode,body:b});});}).on('error',rej);});}
function listenP(s){return new Promise(function(r){s.listen(0,r);});}
