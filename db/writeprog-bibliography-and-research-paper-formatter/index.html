Working implementation

function bibFormat(e,style){style=style||'apa';
if(style==='apa'){var au=e.authors.map(function(a){var p=a.split(' ');return p[p.length-1]+', '+p[0][0]+'.';}).join(', & ');return au+' ('+e.year+'). '+e.title+'. '+e.journal+'.';}
var au=e.authors.map(function(a){var p=a.split(' ');return p[p.length-1]+', '+p.slice(0,-1).join(' ');}).join(', ');
return au+'. "'+e.title+'." '+e.journal+' '+e.volume+' ('+e.year+'): '+e.pages+'.';}
function bibSort(entries){return entries.slice().sort(function(a,b){return a.authors[0].localeCompare(b.authors[0])||a.year-b.year;});}
function bibKey(e){var p=e.authors[0].split(' ');return (p[p.length-1]+e.year).toLowerCase().replace(/[^a-z0-9]/g,'');}
