Working implementation

function hlJs(code){var out='',i=0;var kw={var:1,let:1,const:1,function:1,return:1,if:1,else:1,for:1,while:1,new:1};
function esc(s){return s.replace(/&/g,'&amp;').replace(/</g,'&lt;');}
while(i<code.length){var c=code[i];
if(c==='/'&&code[i+1]==='/'){var j=code.indexOf('\n',i);if(j<0)j=code.length;out+='<span class="cm">'+esc(code.slice(i,j))+'</span>';i=j;continue;}
if(c==='"'||c==="'"){var q=c,j=i+1;while(j<code.length&&code[j]!==q){if(code[j]==='\\')j++;j++;}j++;out+='<span class="st">'+esc(code.slice(i,j))+'</span>';i=j;continue;}
if(/[0-9]/.test(c)){var j=i;while(j<code.length&&/[0-9.]/.test(code[j]))j++;out+='<span class="nm">'+code.slice(i,j)+'</span>';i=j;continue;}
if(/[A-Za-z_$]/.test(c)){var j=i;while(j<code.length&&/[A-Za-z0-9_$]/.test(code[j]))j++;var w=code.slice(i,j);out+=kw[w]?'<span class="kw">'+w+'</span>':esc(w);i=j;continue;}
out+=esc(c);i++;}return out;}
