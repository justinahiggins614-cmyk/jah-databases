function adePlan(request){var steps=[{name:'design',kind:'plan'},{name:'implement',kind:'code'},{name:'test',kind:'test'},{name:'package',kind:'package'}];
return {request:request,steps:steps};}
function adeImplement(step){if(step.kind==='code')return 'function solution(x){ return x*2; }';if(step.kind==='test')return 'assert(solution(21)===42);';return '// '+step.name;}
function adeTest(code,testSrc){try{var v=require('vm').runInNewContext(code+';'+testSrc.replace(/assert\((.*)\);?/,function(m,e){return 'if(!('+e+')) throw new Error("assert failed");';}));return {ok:true};}catch(e){return {ok:false,error:String(e.message)};}}
function adeRun(request){var plan=adePlan(request);var artifacts={};plan.steps.forEach(function(s){artifacts[s.name]=adeImplement(s);});
var t=adeTest(artifacts.implement,artifacts.test);return {plan:plan.steps.map(function(s){return s.name;}),artifacts:artifacts,test:t,ok:t.ok};}
