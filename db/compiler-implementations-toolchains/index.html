JAH-COMP-PUB-000001 — The Compiler Implementations and Toolchains Database — Overview: what a compiler is

ladder rank 1 public

Public edition. Public compiler-construction knowledge, fact-checked reference. A compiler translates source text into an executable form through stages: lexing (characters to tokens), parsing (tokens to a syntax tree), code generation (tree to instructions), and execution (instructions to a result). A toolchain chains these stages with preprocessing, optimization, assembly, and linking. This database carries REAL implementations of every stage — not descriptions of compilers, but compilers. Every demo below executes the actual code. Note: a companion toolchain database (advdb-compiler-language-toolchain) covers toolchain organization; this database is the runnable-implementation companion, and the per-language compiler databases (MiniC, MiniC#, MiniJava, MiniRust, MiniGo) carry the individual language front ends.

Demo: Lexer tokenizes

function cLex(s){var t=[],i=0;while(i<s.length){var c=s[i];if(/\s/.test(c)){i++;continue;}
if(/[0-9]/.test(c)){var j=i;while(j<s.length&&/[0-9]/.test(s[j]))j++;t.push({t:'num',v:+s.slice(i,j)});i=j;continue;}
if('+-*/%()'.indexOf(c)>=0){t.push({t:'op',v:c});i++;continue;}
throw new Error('lexer: bad char '+c);}return t;}
function cParse(t){var p=0;function pk(){return t[p];}function nx(){return t[p++];}
function ex(){var l=tm();while(pk()&&pk().t==='op'&&(pk().v==='+'||pk().v==='-')){var o=nx().v;l={op:o,l:l,r:tm()};}return l;}
function tm(){var l=un();while(pk()&&pk().t==='op'&&(pk().v==='*'||pk().v==='/'||pk().v==='%')){var o=nx().v;l={op:o,l:l,r:un()};}return l;}
function un(){if(pk()&&pk().t==='op'&&pk().v==='-'){nx();return{op:'neg',l:fc()};}return fc();}
function fc(){var x=nx();if(x.t==='num')return{num:x.v};if(x.t==='op'&&x.v==='('){var e=ex();nx();return e;}throw new Error('parser: unexpected');}
var e=ex();if(p!==t.length)throw new Error('parser: trailing tokens');return e;}
function cGen(n,bc){bc=bc||[];if('num' in n){bc.push(['push',n.num]);return bc;}if(n.op==='neg'){cGen(n.l,bc);bc.push(['neg']);return bc;}cGen(n.l,bc);cGen(n.r,bc);bc.push([n.op]);return bc;}
function cRun(bc){var st=[];bc.forEach(function(i){var o=i[0];if(o==='push')st.push(i[1]);else if(o==='neg')st.push(-st.pop());else{var b=st.pop(),a=st.pop();st.push(o==='+'?a+b:o==='-'?a-b:o==='*'?a*b:o==='/'?Math.floor(a/b):a%b);}});return st[0];}
function cCompile(src){return cGen(cParse(cLex(src)));}
function cToolchain(src){var toks=cLex(src),ast=cParse(toks),bc=cGen(ast),val=cRun(bc);return {tokens:toks.length,astNodes:JSON.stringify(ast).length,insns:bc.length,value:val};}
var t=cLex('12 + 34*(5-6)');J({toks:t.map(function(x){return x.t+':'+x.v;})});assert(t.length===9&&t[0].v===12,'lex');

JAH-COMP-PUB-000002 — The Compiler Implementations and Toolchains Database — Stage 1: the lexer

ladder rank 3 public

Public edition. Public compiler-construction knowledge, fact-checked reference. The lexer scans raw characters and emits tokens: numbers, operators, parentheses. It rejects bad characters immediately with a precise error rather than failing mysteriously later. The implementation below is a hand-written scanner — no generator tools, no dependencies — so its behavior is fully visible. The demo tokenizes an expression and asserts the token stream.
