var FIELDS={"id":"JAH-BIZ-000001","company_id":"JAH-BIZ-000001","legal_name":"Northgate Trading Enterprises","title":"Northgate Trading Enterprises","description":"Northgate Trading Enterprises is a 11-50-employee services company based in Zurich, Switzerland, founded in 1997.","industry":"services","city":"Zurich","country":"Switzerland","employee_band":"11-50","website":"https://example-1.com","founded_year":1997};

function show(i){var k=keys[i];if(k===undefined){console.log("no field "+i);return;}console.log("["+i+"] "+k+" ("+(typeof FIELDS[k])+"):");console.log("    "+JSON.stringify(FIELDS[k]).slice(0,300));}
function find(word){word=String(word).toLowerCase();var hits=keys.filter(function(k){return (k+" "+JSON.stringify(FIELDS[k])).toLowerCase().indexOf(word)>=0;});console.log("fields matching \""+word+"\": "+(hits.join(", ")||"(none)"));return hits;}
