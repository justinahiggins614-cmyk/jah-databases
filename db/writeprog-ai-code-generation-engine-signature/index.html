Working implementation

function cgTemplate(name,slots,body){return {name:name,slots:slots,body:body};}
function cgGenerate(tpl,values){var miss=tpl.slots.filter(function(s){return !(s in values);});
if(miss.length)throw new Error('missing slots: '+miss.join(','));var code=tpl.body;
tpl.slots.forEach(function(s){code=code.split('{{'+s+'}}').join(values[s]);});return code;}
function cgLibrary(){return {add:cgTemplate('add',['T'], 'function add(a: {{T}}, b: {{T}}) { return a + b; }'.replace(/: \{\{T\}\}/g,'')),};}
function cgValidate(code){try{new (require('vm').Script)(code);return {ok:true};}catch(e){return {ok:false,error:String(e.message)};}}
