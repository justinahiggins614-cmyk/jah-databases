Working demo program

# Workshop layout checker: zone areas, aisle clearances, circuit loads
ZONES = [('Bench', 72, 30), ('Cutting', 96, 48), ('Assembly', 72, 48), ('Storage', 96, 24)]
AISLES = [42, 48]
CIRCUITS = [('General', 20, 8), ('Lighting', 15, 6)]
total = 0.0
for name, w, d in ZONES:
    area = w * d / 144.0
    total += area
    print("%s: %d x %d in = %.1f sq ft" % (name, w, d, area))
print("Total zone area: %.1f sq ft" % total)
for i, a in enumerate(AISLES, 1):
    print("Aisle %d: %d in clear - %s" % (i, a, "PASS (>=36 in)" if a >= 36 else "FAIL"))
for name, amps, load in CIRCUITS:
    print("%s: %d A load on %d A circuit = %.0f%%" % (name, load, amps, load / amps * 100.0))
