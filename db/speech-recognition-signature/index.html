import hashlib
theta0 = 1.0
target = 2.0
eta = 0.25
def compute():
    loss0 = round((theta0 - target) ** 2, 4)
    grad = 2 * (theta0 - target)
    theta1 = round(theta0 - eta * grad, 4)
    loss1 = round((theta1 - target) ** 2, 4)
    improvement = round(loss0 - loss1, 4)
    out = {"grad": grad, "improvement": improvement, "loss_after": loss1, "loss_before": loss0, "theta_after": theta1, "theta_before": theta0}
    return out
outputs = compute()
result = "|".join(f"{k}={v}" for k, v in sorted(outputs.items()))

outputs2 = compute()
