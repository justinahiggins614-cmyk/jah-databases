Working implementation

function secScan(code){var findings=[];var rules=[
{re:/\beval\s*\(/,sev:'critical',msg:'eval() allows code injection'},
{re:/child_process/,sev:'critical',msg:'process spawning'},
{re:/innerHTML\s*=/,sev:'medium',msg:'innerHTML sink (XSS risk)'},
{re:/(password|passwd|secret|api[_-]?key)\s*[:=]\s*['"][^'"]{3,}['"]/i,sev:'high',msg:'possible hardcoded credential'},
{re:/Math\.random\(\)/,sev:'low',msg:'Math.random is not cryptographic'}];
code.split('\n').forEach(function(l,i){rules.forEach(function(r){if(r.re.test(l))findings.push({line:i+1,sev:r.sev,msg:r.msg});});});
return findings;}
function secGate(findings){return {block:findings.some(function(f){return f.sev==='critical';}),counts:{critical:findings.filter(function(f){return f.sev==='critical';}).length,high:findings.filter(function(f){return f.sev==='high';}).length}};}
