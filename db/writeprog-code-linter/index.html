Working implementation

function lintJs(code){var issues=[];var lines=code.split('\n');
lines.forEach(function(l,i){var n=i+1;var t=l.trim();
if(/(^|[^\w$])var(\s|$)/.test(l))issues.push({line:n,rule:'no-var',msg:'use let/const instead of var'});
if(/[^=!]==[^=]/.test(l))issues.push({line:n,rule:'eqeqeq',msg:'use === instead of =='});
if(t&&!/[{(\[,]$/.test(t)&&!/[;{}]$/.test(t)&&!/^\s*(\/\/|$)/.test(l))issues.push({line:n,rule:'semi',msg:'missing semicolon'});
if(l.length>100)issues.push({line:n,rule:'max-len',msg:'line exceeds 100 chars'});});
return issues;}
