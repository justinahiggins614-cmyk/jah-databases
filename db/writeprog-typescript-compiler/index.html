Working implementation

function tsStrip(src){var s=String(src);
s=s.replace(/interface\s+[A-Za-z_$][\w$]*\s*\{[^}]*\}/g,'');
s=s.replace(/:\s*[A-Za-z_$][\w$]*(<[^>]*>)?(\[\])?/g,function(m){return (/^\s*:/.test(m)&&m.indexOf('?')<0)?'':m;});
s=s.replace(/\b(private|public|readonly)\s+/g,'');
s=s.replace(/\)\s*:\s*[A-Za-z_$][\w$]*(<[^>]*>)?(\[\])?\s*(\{|=>)/g,') $3');
s=s.replace(/as\s+[A-Za-z_$][\w$]*/g,'');
s=s.replace(/<([A-Za-z_$][\w$]*)>/g,'');
return s;}
function tsCheck(src){var js=tsStrip(src);try{new (require('vm').Script)(js);return {ok:true,js:js};}catch(e){return {ok:false,error:String(e.message)};}}

var r=tsCheck('function f(x: number): number { return x*2; }');J({ok:r.ok});assert(r.ok,'parses');

{"ok":true}
