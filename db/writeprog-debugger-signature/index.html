function dbgCompile(src){var toks=src.match(/[A-Za-z_][A-Za-z0-9_]*|[0-9]+|[=+\-*/%;]/g)||[];var prog=[],i=0;
while(i<toks.length){var t=toks[i];
if(/^[A-Za-z_]/.test(t)&&toks[i+1]==='='){var name=t;i+=2;var expr=[];while(i<toks.length&&toks[i]!==';'){expr.push(toks[i]);i++;}i++;prog.push({op:'assign',name:name,expr:expr});}
else if(t==='print'){i++;var expr=[];while(i<toks.length&&toks[i]!==';'){expr.push(toks[i]);i++;}i++;prog.push({op:'print',expr:expr});}
else i++;}
return prog;}
function dbgStep(prog){var state={vars:{},pc:0,out:[]};
function evalExpr(expr){var s=expr.join(' ').replace(/[A-Za-z_][A-Za-z0-9_]*/g,function(v){return '('+(state.vars[v]||0)+')';});return Function('return ('+s+')')();}
return {step:function(){if(state.pc>=prog.length)return {done:true};var ins=prog[state.pc];
if(ins.op==='assign')state.vars[ins.name]=evalExpr(ins.expr);else if(ins.op==='print')state.out.push(evalExpr(ins.expr));
state.pc++;return {done:state.pc>=prog.length,vars:Object.assign({},state.vars),out:state.out.slice()};},
run:function(){var r;do{r=this.step();}while(!r.done);return r;}};}
