var WB_TOOLS={doc:'Word Processor Engine',code:'Universal Code Editor',test:'Automated Code Testing Framework',build:'Build Automation Engine',data:'Spreadsheet Writer'};
function wbRoute(request){var t=request.toLowerCase();if(/document|write|essay/.test(t))return 'doc';if(/test/.test(t))return 'test';if(/build|deploy/.test(t))return 'build';if(/data|sheet|csv/.test(t))return 'data';return 'code';}
function wbPipeline(request){var route=wbRoute(request);var stages=['plan','generate','verify','deliver'];var log=stages.map(function(s){return s+':'+WB_TOOLS[route];});
return {request:request,tool:WB_TOOLS[route],stages:stages,log:log};}
function wbDeliver(pipe,artifact){return {tool:pipe.tool,artifact:artifact,delivered:true,at:new Date(0).toISOString()};}
