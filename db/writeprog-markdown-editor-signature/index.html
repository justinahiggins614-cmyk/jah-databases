Working implementation

function mdToHtml(md){var h=String(md).replace(/&/g,'&amp;').replace(/</g,'&lt;');
h=h.replace(/^### (.*)$/gm,'<h3>$1</h3>').replace(/^## (.*)$/gm,'<h2>$1</h2>').replace(/^# (.*)$/gm,'<h1>$1</h1>');
h=h.replace(/\*\*(.+?)\*\*/g,'<strong>$1</strong>').replace(/\*(.+?)\*/g,'<em>$1</em>').replace(/`(.+?)`/g,'<code>$1</code>');
h=h.replace(/\[([^\]]+)\]\(([^)]+)\)/g,'<a href="$2">$1</a>');
var lines=h.split('\n'),out=[],inList=false;
lines.forEach(function(l){if(/^- /.test(l)){if(!inList){out.push('<ul>');inList=true;}out.push('<li>'+l.slice(2)+'</li>');}
else{if(inList){out.push('</ul>');inList=false;}if(/^<h[123]>/.test(l)||!l.trim())out.push(l);else out.push('<p>'+l+'</p>');}});
if(inList)out.push('</ul>');return out.join('\n');}
function mdToc(md){var t=[];String(md).split('\n').forEach(function(l){var m=l.match(/^(#{1,3}) (.*)/);if(m)t.push({level:m[1].length,title:m[2]});});return t;}
