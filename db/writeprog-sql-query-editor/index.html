Working implementation

function sqlTokenize(q){var re=/('(?:[^']|'')*'|[A-Za-z_][A-Za-z0-9_]*|[0-9]+(?:\.[0-9]+)?|<=|>=|<>|!=|[=<>(),;.*])/g;var m,t=[];while((m=re.exec(q)))t.push(m[0]);return t;}
function sqlValidateSelect(q){var t=sqlTokenize(q.toUpperCase());var errs=[];
if(t[0]!=='SELECT')errs.push('must start with SELECT');
var fi=t.indexOf('FROM');if(fi<0)errs.push('missing FROM');
if(t[t.length-1]!==';')errs.push('missing trailing semicolon');
var cols=t.slice(1,fi).filter(function(x){return x!==',';});
if(cols.length===0)errs.push('no columns listed');
return {ok:errs.length===0,errors:errs,columns:t[1]==='*'?['*']:cols,table:fi>=0?t[fi+1]:null};}
function sqlHighlight(q){return sqlTokenize(q).map(function(t){return /^(SELECT|FROM|WHERE|ORDER|BY|GROUP|JOIN|ON|AND|OR|INSERT|UPDATE|DELETE)$/.test(t.toUpperCase())?'<b>'+t+'</b>':t;}).join(' ');}
