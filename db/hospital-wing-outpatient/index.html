def no_show_risk(prior_no_shows, days_since_last_visit, distance_miles):
    """Simple no-show risk flag for overbooking decisions."""
    score = prior_no_shows * 2 + (1 if days_since_last_visit > 365 else 0) + (1 if distance_miles > 20 else 0)
    return f"risk score {score} -> {'HIGH: double-book + reminder call' if score >= 3 else 'standard reminder'}"
