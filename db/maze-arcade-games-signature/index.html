Amber Orchard Maze Rank 1 · simple · maze · beginner

Amber Orchard Maze is an original maze game about a harvester gathering amber apples through orchard lanes at sunset. The lanes wind around apple crates and old fences, and a watchful orchard dog patrols the far lanes, always trotting toward the sound of rustling leaves. Every amber apple adds points, and the harvest is complete when the orchard is bare.

The rules are simple: move through the lanes, step on amber apples for ten points each, and avoid the dog, who costs a life and sends you back to the orchard gate if you meet. You have three lives, and the orchard is cleared when every apple is gathered.

Orchard harvesters clear the near rows first and save the far corner for last, when the dog's patrol is predictable. Never enter a dead-end lane while the dog is close, and always leave a second way out. The orchard rewards a calm, planned harvest far more than a hurried dash.

import sys
PARAMS = {'map': ['###########', '#P........#', '#..#......#', '#.......#.#', '#....#....#', '#........G#', '###########']}
AUTO = ['right', 'right', 'right', 'right', 'down', 'down', 'left', 'left', 'down', 'right', 'right', 'right']

def parse_map():
    walls, dots, player, ghost = set(), set(), None, None
    for y, row in enumerate(PARAMS["map"]):
        for x, ch in enumerate(row):
            if ch == "#":
                walls.add((x, y))
            elif ch == ".":
                dots.add((x, y))
            elif ch == "P":
                player = (x, y)
            elif ch == "G":
                ghost = (x, y)
    return walls, dots, player, ghost

DIRS = {"up": (0, -1), "down": (0, 1), "left": (-1, 0), "right": (1, 0)}

def step(state, cmd):
    walls, dots, player, ghost, score, lives = state
    dx, dy = DIRS.get(cmd, (0, 0))
    nx, ny = player[0] + dx, player[1] + dy
    if (nx, ny) not in walls:
        player = (nx, ny)
    if player in dots:
        dots = dots - {player}
        score += 10
    if ghost is not None:
        gx, gy = ghost
        best, bestd = ghost, abs(gx - player[0]) + abs(gy - player[1])
        for mx, my in [(1, 0), (-1, 0), (0, 1), (0, -1)]:
            cand = (gx + mx, gy + my)
            if cand in walls:
                continue
            d = abs(cand[0] - player[0]) + abs(cand[1] - player[1])
            if d < bestd:
                best, bestd = cand, d
        ghost = best
    if ghost is not None and player == ghost:
        lives -= 1
        walls, dots, player, ghost = reset_positions()
    return (walls, dots, player, ghost, score, lives)

def reset_positions():
    walls, dots, player, ghost = parse_map()
    return walls, dots, player, ghost

def result_of(state):
    walls, dots, player, ghost, score, lives = state
    if not dots:
        return "WIN"
    if lives <= 0:
        return "OUT OF LIVES"
    return "ONGOING"

def auto_play():
    walls, dots, player, ghost = parse_map()
    state = (walls, dots, player, ghost, 0, 3)
    print("Maze tribute: auto session")
    for i, cmd in enumerate(AUTO):
        state = step(state, cmd)
        walls, dots, player, ghost, score, lives = state
        print("Step %d: %s -> pos %s score %d dots %d lives %d" % (i + 1, cmd, player, score, len(dots), lives))
        if result_of(state) != "ONGOING":
            break
    print("FINAL SCORE: %d" % state[4])
    print("RESULT: %s" % result_of(state))

def play():
    walls, dots, player, ghost = parse_map()
    state = (walls, dots, player, ghost, 0, 3)
    print("Maze tribute: type up/down/left/right, or quit. Eat every dot, avoid the ghost.")
    for line in sys.stdin:
        cmd = line.strip().lower()
        if cmd == "quit":
            break
        if cmd not in DIRS:
            print("Unknown command: %s" % cmd)
            continue
        state = step(state, cmd)
        walls, dots, player, ghost, score, lives = state

    print("RESULT: %s" % result_of(state))

if __name__ == "__main__":
    if "--play" in sys.argv:
        play()
    else:
