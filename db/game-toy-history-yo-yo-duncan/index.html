Pinball physics + payout math

Simulates a bouncing pinball's energy loss and computes slot-machine expected value — the math of the midway. Save as demo.py and run python3 demo.py.

# Pinball bounce + slot EV (public-domain game math)
import math, random

def bounce(v0, restitution=0.7, bounces=6):
    v, total = v0, 0.0
    for _ in range(bounces):
        total += v
        v *= restitution
    return total

print(f"ball at 3 m/s, e=0.7: total travel ~{bounce(3):.1f} m over 6 bounces")

# Slot machine EV: 3 reels, Liberty-Bell-ish paytable (simplified)
random.seed(7)
SYMBOLS = ["bell", "horseshoe", "star", "diamond", "heart", "spade"]
PAYS = {("bell",)*3: 20, ("horseshoe",)*3: 10}
def spin():
    return tuple(random.choice(SYMBOLS) for _ in range(3))
N, won = 200000, 0
for _ in range(N):
    won += PAYS.get(spin(), 0)
print(f"slot EV: {won/N:.3f} coins per 1-coin bet (house edge {100*(1-won/N):.1f}%)")
