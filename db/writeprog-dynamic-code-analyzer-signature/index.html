Working implementation

function daTrace(fn,name){var calls=[];
var wrapped=function(){var args=Array.prototype.slice.call(arguments);var t0=Date.now();
try{var r=fn.apply(this,args);calls.push({name:name,args:args,ok:true,ms:Date.now()-t0});return r;}
catch(e){calls.push({name:name,args:args,ok:false,error:String(e.message),ms:Date.now()-t0});throw e;}};
wrapped.calls=function(){return calls;};wrapped.callCount=function(){return calls.length;};return wrapped;}
function daReport(traced){var cs=traced.calls();var fails=cs.filter(function(c){return !c.ok;});
return {calls:cs.length,failures:fails.length,avgMs:cs.length?+(cs.reduce(function(a,c){return a+c.ms;},0)/cs.length).toFixed(2):0};}
