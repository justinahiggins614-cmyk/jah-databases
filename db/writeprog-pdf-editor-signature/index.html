Working implementation

function pdfMake(lines){var content='BT /F1 18 Tf 50 750 Td 22 TL ';
content+=lines.map(function(l){return '('+l.replace(/\\/g,'\\\\').replace(/\(/g,'\\(').replace(/\)/g,'\\)')+') Tj T*';}).join(' ')+' ET';
var objs=[];objs[1]='<< /Type /Catalog /Pages 2 0 R >>';objs[2]='<< /Type /Pages /Kids [3 0 R] /Count 1 >>';
objs[3]='<< /Type /Page /Parent 2 0 R /MediaBox [0 0 612 792] /Contents 4 0 R /Resources << /Font << /F1 5 0 R >> >> >>';
objs[4]='<< /Length '+content.length+' >>\nstream\n'+content+'\nendstream';
objs[5]='<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica >>';
var pdf='%PDF-1.4\n',off=[0];
for(var i=1;i<=5;i++){off[i]=pdf.length;pdf+=i+' 0 obj\n'+objs[i]+'\nendobj\n';}
var xref=pdf.length;pdf+='xref\n0 6\n0000000000 65535 f \n';
for(var j=1;j<=5;j++){var s=String(off[j]);while(s.length<10)s='0'+s;pdf+=s+' 00000 n \n';}
pdf+='trailer\n<< /Size 6 /Root 1 0 R >>\nstartxref\n'+xref+'\n%%EOF';return pdf;}function pdfTextItems(pdf){var out=[],re=/\((?:\\.|[^()\\])*\)\s*Tj/g,m;while((m=re.exec(pdf))){out.push(m[0].slice(1,m[0].lastIndexOf(')')).replace(/\\(.)/g,'$1'));}return out;}
function pdfReplace(pdf,find,repl){return pdf.split('('+find+') Tj').join('('+repl+') Tj');}
function pdfPageCount(pdf){var m=pdf.match(/\/Count\s+(\d+)/);return m?parseInt(m[1],10):0;}
