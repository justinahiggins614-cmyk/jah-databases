Working implementation

function xmlWellFormed(s){var stack=[];var re=/<\/?([A-Za-z_][\w.-]*)[^>]*?(\/?)>/g;var m;
while((m=re.exec(s))){var tag=m[1],selfClose=m[2]==='/'||m[0].slice(-2)==='/>';var closing=m[0][1]==='/';
if(selfClose)continue;if(closing){if(stack.length===0||stack.pop()!==tag)return {ok:false,error:'mismatched close </'+tag+'>'};}
else stack.push(tag);}
return stack.length===0?{ok:true}:{ok:false,error:'unclosed <'+stack[stack.length-1]+'>'};}
function xmlFind(s,tag){var re=new RegExp('<'+tag+'(?:\\s[^>]*)?>([\\s\\S]*?)</'+tag+'>','g');var out=[],m;while((m=re.exec(s)))out.push(m[1]);return out;}
function xmlEscape(s){return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');}
