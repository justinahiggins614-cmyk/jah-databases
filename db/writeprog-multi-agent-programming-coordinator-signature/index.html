Working implementation

function macSplit(project){return project.tasks.map(function(t,i){return {id:'agent-'+i,task:t};});}
function macWorker(assign,skills){var t=assign.task;
if(skills.indexOf(t.kind)<0)return {id:assign.id,ok:false,error:'no skill for '+t.kind};
var result=t.kind==='code'?'// code for '+t.name:t.kind==='test'?'assert('+t.name+'());':'docs for '+t.name;
return {id:assign.id,ok:true,kind:t.kind,result:result};}
function macCoordinate(project,skills){var assigns=macSplit(project);var results=assigns.map(function(a){return macWorker(a,skills);});
var failed=results.filter(function(r){return !r.ok;});return {agents:assigns.length,results:results,ok:failed.length===0,failed:failed};}
