Working implementation

function citeApa(e){var au=e.authors.map(function(a){var p=a.split(' ');return p[p.length-1]+', '+p[0].charAt(0)+'.';}).join(', & ');
return au+' ('+e.year+'). '+e.title+'. '+e.source+'.';}
function citeMla(e){var au=e.authors.map(function(a){var p=a.split(' ');return p[p.length-1]+', '+p.slice(0,-1).join(' ');}).join(', ');
return au+'. "'+e.title+'." '+e.source+', '+e.year+'.';}
function citeValidate(e){var miss=[];['authors','year','title','source'].forEach(function(k){if(!e[k]||(k==='authors'&&!e[k].length))miss.push(k);});return {ok:miss.length===0,missing:miss};}
