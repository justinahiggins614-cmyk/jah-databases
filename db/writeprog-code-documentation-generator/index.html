Working implementation

function docGen(code){var docs=[];var re=/\/\*\*([\s\S]*?)\*\/\s*(?:function\s+([A-Za-z_$][\w$]*)|(?:const|let|var)\s+([A-Za-z_$][\w$]*)\s*=\s*(?:function|\())/g;var m;
while((m=re.exec(code))){var name=m[2]||m[3];var desc=m[1].replace(/^\s*\*\s?/gm,'').trim();docs.push({name:name,doc:desc});}
return docs;}
function docMarkdown(docs){return docs.map(function(d){return '### '+d.name+'\n\n'+d.doc+'\n';}).join('\n');}

Demo: Markdown rendering

var md=docMarkdown([{name:'add',doc:'Adds.'}]);J({md:md});assert(md.indexOf('### add')===0,'md');

Verified output (2026-10-21, v24.20.0):

{"md":"### add\n\nAdds.\n"}
