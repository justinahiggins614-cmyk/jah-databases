def das28(tender28, swollen28, esr, patient_global):
    """DAS28 rheumatoid arthritis activity estimator (simplified)."""
    import math
    d = 0.56*math.sqrt(tender28) + 0.28*math.sqrt(swollen28) + 0.70*math.log(esr) + 0.014*patient_global
    band = "remission" if d < 2.6 else "low" if d < 3.2 else "moderate" if d <= 5.1 else "high"
    return f"DAS28 {d:.2f} -> {band} disease activity"
