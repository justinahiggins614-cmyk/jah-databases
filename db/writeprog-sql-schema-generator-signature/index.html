Working implementation

function ddlTable(t){var cols=t.columns.map(function(c){var s='  '+c.name+' '+c.type;if(c.pk)s+=' PRIMARY KEY';if(c.notNull)s+=' NOT NULL';if(c.unique)s+=' UNIQUE';if(c.def!==undefined)s+=" DEFAULT '"+c.def+"'";return s;});
var fks=(t.foreignKeys||[]).map(function(f){return '  FOREIGN KEY ('+f.col+') REFERENCES '+f.refTable+'('+f.refCol+')';});
return 'CREATE TABLE '+t.name+' (\n'+cols.concat(fks).join(',\n')+'\n);';}
function ddlIndex(table,col){return 'CREATE INDEX idx_'+table+'_'+col+' ON '+table+'('+col+');';}
function ddlSchema(tables){return tables.map(ddlTable).join('\n\n');}
