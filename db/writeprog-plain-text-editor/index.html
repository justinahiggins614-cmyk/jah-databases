Working implementation

function txBuf(t){return {text:t||''};}
function txInsert(b,pos,s){b.text=b.text.slice(0,pos)+s+b.text.slice(pos);return b;}
function txDelete(b,pos,n){b.text=b.text.slice(0,pos)+b.text.slice(pos+n);return b;}
function txFindAll(b,pat){var r=[],i=-1;while((i=b.text.indexOf(pat,i+1))>=0)r.push(i);return r;}
function txLines(b){return b.text.split('\n').length;}
function txReplaceAll(b,find,repl){b.text=b.text.split(find).join(repl);return b;}

Demo: Find all occurrences

var b=txBuf('one two one');var f=txFindAll(b,'one');J({found:f});assert(f.length===2&&f[0]===0&&f[1]===8,'find');
