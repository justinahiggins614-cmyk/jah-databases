Working implementation

function tsBreak(paragraph,width){var words=paragraph.split(/\s+/).filter(Boolean);var lines=[],cur='';
words.forEach(function(w){if(cur.trim()&&(cur+' '+w).trim().length>width){lines.push(cur.trim());cur=w;}else cur=(cur+' '+w).trim();});
if(cur.trim())lines.push(cur.trim());return lines;}
function tsJustify(line,width){var ws=line.split(' ');if(ws.length<2)return line;var gaps=ws.length-1;var total=ws.join('').length;var need=width-total;var per=Math.floor(need/gaps),extra=need%gaps;var out=ws[0];
for(var i=1;i<ws.length;i++){var sp=per+(i<=extra?1:0);for(var s=0;s<sp;s++)out+=' ';out+=ws[i];}return out;}
function tsRaggedRight(paras,width){return paras.map(function(p){return tsBreak(p,width);});}
