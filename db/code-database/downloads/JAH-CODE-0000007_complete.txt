description: A hash map with separate chaining stores colliding keys in per-bucket lists. Simple, correct, and the classic teaching implementation.
code: class HM:
    def __init__(self, n=8):
        self.b = [[] for _ in range(n)]
    def _i(self, k):
        return hash(k) % len(self.b)
    def put(self, k, v):
        b = self.b[self._i(k)]
        for i, (kk, _) in enumerate(b):
            if kk == k:
                b[i] = (k, v)
                return
        b.append((k, v))
    def get(self, k):
        for kk, v in self.b[self._i(k)]:
            if kk == k:
                return v

m = HM()
m.put("a", 1); m.put("a", 2)
