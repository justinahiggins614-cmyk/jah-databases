code: class Sub:
    def __init__(self):
        self.o = []
    def attach(self, fn):
        self.o.append(fn)
    def notify(self, v):
        for fn in self.o:
            fn(v)

s, log = Sub(), []
s.attach(lambda v: log.append("a" + v))
s.attach(lambda v: log.append("b" + v))
s.notify("!")
print("".join(log))
