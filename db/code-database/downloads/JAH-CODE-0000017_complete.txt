code: def merge(iv):
    iv.sort()
    o = []
    for s, e in iv:
        if o and s <= o[-1][1]:
            o[-1][1] = max(o[-1][1], e)
        else:
            o.append([s, e])
    return o

print(merge([[1, 3], [2, 6], [8, 10], [15, 18]]))
