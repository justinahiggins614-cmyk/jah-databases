code: from collections import deque
def bfs(g, s, t):
    q, seen = deque([(s, 0)]), {s}
    while q:
        n, d = q.popleft()
        if n == t:
            return d
        for m in g.get(n, []):
            if m not in seen:
                seen.add(m); q.append((m, d + 1))
    return -1

g = {"a": ["b", "c"], "b": ["d"], "c": ["d"], "d": []}
print(bfs(g, "a", "d"))
