function stwGenerate(prompt,attempt){var drafts=['Draft: '+prompt,'Revised draft: '+prompt+' (clearer)','Final draft: '+prompt+' (clear, concise, checked)'];return drafts[Math.min(attempt,drafts.length-1)];}
function stwChecks(){return [
{name:'non-empty',fn:function(t){return t.trim().length>0;}},
{name:'ends-punct',fn:function(t){return /[.!?]$/.test(t.trim());}},
{name:'no-double-space',fn:function(t){return t.indexOf('  ')<0;}},
{name:'min-words',fn:function(t){return t.split(/\s+/).length>=3;}}];}
function stwTest(text){return stwChecks().map(function(c){return {name:c.name,pass:c.fn(text)};});}
function stwLoop(prompt){var attempt=0,text='';while(attempt<5){text=stwGenerate(prompt,attempt)+'.';var res=stwTest(text);
if(res.every(function(r){return r.pass;}))return {text:text,attempts:attempt+1,checks:res};attempt++;}return {text:text,attempts:attempt,checks:stwTest(text),gaveUp:true};}
