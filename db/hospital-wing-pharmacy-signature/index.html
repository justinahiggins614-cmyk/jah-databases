def renal_dose_adjust(creatinine, age, weight_kg, female, normal_dose_mg):
    """Cockcroft-Gault creatinine clearance -> dose band suggestion."""
    scr = max(creatinine, 0.6)
    crcl = ((140 - age) * weight_kg) / (72 * scr)
    if female: crcl *= 0.85
    factor = 1.0 if crcl >= 60 else 0.75 if crcl >= 30 else 0.5
    return f"CrCl ~{crcl:.0f} mL/min -> give {round(normal_dose_mg*factor)} mg (verify with pharmacist)"
print(renal_dose_adjust(1.6, 72, 70, False, 1000))

    """Cockcroft-Gault creatinine clearance -> dose band suggestion."""
    scr = max(creatinine, 0.6)
    crcl = ((140 - age) * weight_kg) / (72 * scr)
    if female: crcl *= 0.85
    factor = 1.0 if crcl >= 60 else 0.75 if crcl >= 30 else 0.5
    return f"CrCl ~{crcl:.0f} mL/min -> give {round(normal_dose_mg*factor)} mg (verify with pharmacist)"
print(renal_dose_adjust(1.6, 72, 70, False, 1000))
