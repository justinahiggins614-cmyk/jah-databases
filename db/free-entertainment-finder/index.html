design_title = 'Free Entertainment Finder Field Guide Index'
design = 'A free entertainment finder website: a visitor picks a generic kind of free entertainment, such as a puppet show in a park or a story hour in a library, and the site shows illustrative sample listings of that kind, each with a plain description of what the evening holds, how long it lasts, and where it happens. The site names no real performer, copies no real listing, and makes no claim that any sample entry is a real event. This record is the field guide index for the whole database: it lists all twenty-three content records in ladder order, simple to compounded to complex, and tells a reader where to start.'
steps = ['Add a tournament bracket with rounds, divisions, and a season', 'Add a creator studio so visitors can build a new free entertainment listing', 'Add a review step so every new free entertainment listing is checked before it is published', 'Add the free entertainment listing in many languages, with the same rules in each', 'Add an offline pack so the free entertainment listing works with no connection']

print("Count: %d" % count)
