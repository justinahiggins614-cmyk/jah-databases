Beatmania Rank 1 · simple · rhythm arcade · beginner

Beatmania, released in 1997, is a disc-jockey game in which the player scratches a turntable and taps keys in time with the music, building a groove from the song's beat track.

The tribute demo is a sixteen-beat excerpt, not the full song chart: sixteen beats scroll by, and the player presses the matching key for each one. A hit builds the combo and adds bonus points, while a miss breaks the combo. The demo is an original program inspired by the mechanics, not the arcade game itself.

Good rhythm play is about staying loose through the misses. A broken combo is not a failed song, and the next beat is always a fresh chance to rebuild. Read two beats ahead, keep the hands relaxed, and let the pattern carry you.

Playable tribute note: The demo below is an original tribute inspired by the mechanics of Beatmania (1997). It is not Beatmania (1997), and it uses no art, music, or code from that game.

How to play

Type the key for each beat: left, up, down, or right; quit to stop.

Working demo program

Save the code below as game.py and run python3 game.py --play to play it. Running it without arguments replays the scripted session whose output is stored with the record.

import sys
PARAMS = {'chart': ['right', 'left', 'right', 'down', 'left', 'right', 'up', 'right', 'up', 'up', 'down', 'right', 'up', 'down', 'up', 'right']}
AUTO = ['down', 'left', 'right', 'down', 'left', 'right', 'up', 'right', 'up', 'up', 'down', 'right', 'up', 'down', 'up', 'right']

def auto_play():
    score, combo, maxcombo, hits = 0, 0, 0, 0
    print("Rhythm tribute: auto session")
    for i, key in enumerate(PARAMS["chart"]):
        pressed = AUTO[i] if i < len(AUTO) else "miss"
        hit = (pressed == key)
        if hit:
            combo += 1
            maxcombo = max(maxcombo, combo)
            score += 100 + combo * 10
            hits += 1
        else:
            combo = 0
        print("Beat %d: need %s pressed %s %s score %d" % (i + 1, key, pressed, "HIT" if hit else "MISS", score))
    print("FINAL SCORE: %d" % score)
    print("RESULT: HITS %d/%d MAX COMBO %d" % (hits, len(PARAMS["chart"]), maxcombo))

def play():
    score, combo, maxcombo, hits = 0, 0, 0, 0
    print("Rhythm tribute: type the key for each beat (left/up/down/right), or quit.")
    for i, key in enumerate(PARAMS["chart"]):
        print("Beat %d: press a key." % (i + 1))
        pressed = None
        for line in sys.stdin:
            cmd = line.strip().lower()
            if cmd == "quit":
                print("FINAL SCORE: %d" % score)
                print("RESULT: HITS %d/%d MAX COMBO %d" % (hits, len(PARAMS["chart"]), maxcombo))
                return
            pressed = cmd
            break
        hit = (pressed == key)
        if hit:
            combo += 1
            maxcombo = max(maxcombo, combo)
            score += 100 + combo * 10
            hits += 1
        else:
            combo = 0
        print("%s score %d" % ("HIT" if hit else "MISS", score))
    print("FINAL SCORE: %d" % score)
    print("RESULT: HITS %d/%d MAX COMBO %d" % (hits, len(PARAMS["chart"]), maxcombo))

if __name__ == "__main__":
    if "--play" in sys.argv:
        play()
    else:
        auto_play()

AUTO = ['right', 'down', 'right', 'right', 'down', 'down', 'down', 'right', 'up', 'left', 'right', 'right', 'down', 'down', 'up', 'down']

def auto_play():
    score, combo, maxcombo, hits = 0, 0, 0, 0
    print("Rhythm tribute: auto session")
    for i, key in enumerate(PARAMS["chart"]):
        pressed = AUTO[i] if i < len(AUTO) else "miss"
        hit = (pressed == key)
        if hit:
            combo += 1
            maxcombo = max(maxcombo, combo)
            score += 100 + combo * 10
            hits += 1
        else:
            combo = 0
        print("Beat %d: need %s pressed %s %s score %d" % (i + 1, key, pressed, "HIT" if hit else "MISS", score))
    print("FINAL SCORE: %d" % score)
    print("RESULT: HITS %d/%d MAX COMBO %d" % (hits, len(PARAMS["chart"]), maxcombo))

def play():
    score, combo, maxcombo, hits = 0, 0, 0, 0
    print("Rhythm tribute: type the key for each beat (left/up/down/right), or quit.")
    for i, key in enumerate(PARAMS["chart"]):
        print("Beat %d: press a key." % (i + 1))
        pressed = None
        for line in sys.stdin:
            cmd = line.strip().lower()
            if cmd == "quit":
                print("FINAL SCORE: %d" % score)
                print("RESULT: HITS %d/%d MAX COMBO %d" % (hits, len(PARAMS["chart"]), maxcombo))
                return
            pressed = cmd
            break
        hit = (pressed == key)
        if hit:
            combo += 1
            maxcombo = max(maxcombo, combo)
            score += 100 + combo * 10
            hits += 1
        else:
            combo = 0
        print("%s score %d" % ("HIT" if hit else "MISS", score))
    print("FINAL SCORE: %d" % score)
    print("RESULT: HITS %d/%d MAX COMBO %d" % (hits, len(PARAMS["chart"]), maxcombo))

if __name__ == "__main__":
    if "--play" in sys.argv:
        play()
    else:
        auto_play()
