Working implementation

var PARA_SYN={quick:['fast','rapid'],brown:['tan','umber'],happy:['glad','joyful'],big:['large','huge'],good:['fine','great'],said:['stated','remarked'],very:['highly','extremely'],use:['utilize','employ']};
function paraphrase(text,seed){var words=text.split(/(\s+)/);var out=[];var n=seed||0;
words.forEach(function(w){var k=w.toLowerCase();if(PARA_SYN[k]&&(n%2===0)){var syn=PARA_SYN[k][n%PARA_SYN[k].length];out.push(/^[A-Z]/.test(w)?syn.charAt(0).toUpperCase()+syn.slice(1):syn);}else out.push(w);n++;});
return out.join('');}
function paraphraseVariants(text){var v=[];for(var s=0;s<3;s++)v.push(paraphrase(text,s));return v;}
