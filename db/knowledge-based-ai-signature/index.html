import hashlib
items = [('trend_watcher', 6), ('prototype_bench', 10), ('failure_lister', 8), ('horizon_planner', 7), ('ethics_reviewer', 5)]
def compute():
    total = sum(v for _, v in items)
    count = len(items)
    mean_value = round(total / count, 4) if count else 0
    smallest = min(v for _, v in items)
    largest = max(v for _, v in items)
    smallest_key = min(items, key=lambda kv: kv[1])[0]
    largest_key = max(items, key=lambda kv: kv[1])[0]
    out = {"count": count, "largest": largest, "largest_key": largest_key, "mean_value": mean_value, "smallest": smallest, "smallest_key": smallest_key, "total": total}
    return out
outputs = compute()
result = "|".join(f"{k}={v}" for k, v in sorted(outputs.items()))
h1 = hashlib.sha256(result.encode()).hexdigest()
outputs2 = compute()
result2 = "|".join(f"{k}={v}" for k, v in sorted(outputs.items()))
h2 = hashlib.sha256(result2.encode()).hexdigest()
for k, v in sorted(outputs.items()):
    print(f"{k}: {v}")
print(f"Result: {result}")
print(f"Run 1 SHA-256: {h1}")
print(f"Run 2 SHA-256: {h2}")
