Working implementation

function cssRule(sel,decls){var body=Object.keys(decls).map(function(k){return '  '+k+': '+decls[k]+';';}).join('\n');return sel+' {\n'+body+'\n}';}
function cssSheet(rules){return rules.map(function(r){return cssRule(r.sel,r.decls);}).join('\n\n');}
function cssMedia(query,rules){return '@media '+query+' {\n'+cssSheet(rules).split('\n').map(function(l){return '  '+l;}).join('\n')+'\n}';}

Demo: Media query wraps

var m=cssMedia('(max-width:600px)',[{sel:'p',decls:{'font-size':'14px'}}]);J({m:m});assert(m.indexOf('@media (max-width:600px)')===0,'media');

Verified output (2026-10-21, v24.20.0):

{"m":"@media (max-width:600px) {\n  p {\n    font-size: 14px;\n  }\n}"}
