Working implementation

function archDesign(reqs){var comps=[];var seen={};
function add(name,why){if(!seen[name]){seen[name]=1;comps.push({name:name,reason:why});}}
reqs.forEach(function(r){var t=r.toLowerCase();
if(/stor|data|persist/.test(t))add('Database','stores '+r);
if(/user|login|auth/.test(t))add('Auth Service','handles '+r);
if(/api|endpoint|rest/.test(t))add('API Gateway','exposes '+r);
if(/ui|page|display|dashboard/.test(t))add('Frontend App','renders '+r);
if(/realtime|stream|socket/.test(t))add('Event Bus','streams '+r);
if(/report|analytics/.test(t))add('Analytics Worker','computes '+r);});
if(comps.length===0)add('Monolith App','simple requirements');
return {components:comps};}
function archDiagram(d){var w=d.components.map(function(c){return c.name;});var width=Math.max.apply(null,w.map(function(x){return x.length;}));
var line='+'+('-'.repeat(width+2))+'+';
return [line].concat(d.components.map(function(c){return '| '+c.name+(' '.repeat(width-c.name.length))+' |';})).concat([line]).join('\n');}
