Working implementation

function adsExtract(code){var fns=[];var re=/function\s+([A-Za-z_$][\w$]*)\s*\(([^)]*)\)/g;var m;
while((m=re.exec(code)))fns.push({name:m[1],params:m[2].split(',').map(function(s){return s.trim();}).filter(Boolean)});
return fns;}
function adsSync(code,oldDoc){var fns=adsExtract(code);var lines=['# API Reference',''];
fns.forEach(function(f){lines.push('## '+f.name+'('+f.params.join(', ')+')','');});
var fresh=lines.join('\n');return {doc:fresh,added:fns.filter(function(f){return oldDoc.indexOf('## '+f.name)!==0&&oldDoc.indexOf('\n## '+f.name)<0;}).map(function(f){return f.name;}),removed:[]};}
function adsStaleCheck(code,doc){var fns=adsExtract(code);return fns.filter(function(f){return doc.indexOf(f.name)<0;}).map(function(f){return f.name;});}
