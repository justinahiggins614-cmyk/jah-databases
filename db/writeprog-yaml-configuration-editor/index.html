Working implementation

function yamlParse(s){var root={},stack=[{obj:root,indent:-1}];s.split('\n').forEach(function(line){
if(!line.trim()||line.trim()[0]==='#')return;var indent=line.match(/^\s*/)[0].length;var t=line.trim();
while(stack.length>1&&indent<=stack[stack.length-1].indent)stack.pop();var cur=stack[stack.length-1].obj;
if(t[0]==='-'){var v=yamlVal(t.slice(1).trim());if(!Array.isArray(cur._list))cur._list=[];cur._list.push(v);return;}
var m=t.match(/^([^:]+):\s*(.*)$/);if(!m)return;var k=m[1].trim(),v=yamlVal(m[2]);
if(v===null&&m[2]===''){var child={};cur[k]=child;stack.push({obj:child,indent:indent});}else cur[k]=v;});
(function clean(o){Object.keys(o).forEach(function(k){if(k==='_list'){return;}if(o[k]&&typeof o[k]==='object'){if(o[k]._list!==undefined&&Object.keys(o[k]).length===1)o[k]=o[k]._list;else clean(o[k]);}});})(root);
return root;}
function yamlVal(v){if(v==='' )return null;if(v==='true')return true;if(v==='false')return false;if(v==='null')return null;
if(/^-?\d+$/.test(v))return parseInt(v,10);if(/^-?\d*\.\d+$/.test(v))return parseFloat(v);
if((v[0]==='"'&&v.slice(-1)==='"')||(v[0]==="'"&&v.slice(-1)==="'"))return v.slice(1,-1);return v;}
function yamlEmit(o,ind){ind=ind||0;var pad='';for(var i=0;i<ind;i++)pad+='  ';var out=[];
Object.keys(o).forEach(function(k){var v=o[k];
if(v&&typeof v==='object'&&!Array.isArray(v)){out.push(pad+k+':');out.push(yamlEmit(v,ind+1));}
else if(Array.isArray(v)){out.push(pad+k+':');v.forEach(function(x){out.push(pad+'- '+x);});}
else out.push(pad+k+': '+v);});return out.join('\n');}
