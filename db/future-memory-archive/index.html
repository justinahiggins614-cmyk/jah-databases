design_title = 'Future Memory Archive Field Guide Index'
design = 'A memory archive website: a visitor picks a kind of sealed record, such as a letter to the future or a note for a birthday, and the archive shows the vault in plain words, when it may be opened, and who it is for. Every vault is a generic design study, and the archive never claims to hold any real letter. This record is the field guide index for the whole database: it lists all twenty-three content records in ladder order, simple to compounded to complex, and tells a reader where to start.'
steps = ['Add a tournament bracket with rounds, divisions, and a season', 'Add a creator studio so visitors can build a new memory vault', 'Add a review step so every new memory vault is checked before it is published', 'Add the memory vault in many languages, with the same rules in each', 'Add an offline pack so the memory vault works with no connection']
count = 23
each = 1
total = count * each
print("Website design study: " + design_title)
print("Design (a design study, not a live website): " + design)
print("Note: this program simulates the design on paper; it is not a test on a live website.")
for i, step in enumerate(steps, 1):
    print("Step %d: %s" % (i, step))
print("Count: %d" % count)
print("Each: %d" % each)
print("Total: %d" % total)
