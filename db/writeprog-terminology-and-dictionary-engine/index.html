Working implementation

function termDb(){return {terms:{}};}
function termAdd(db,term,defn,domain){db.terms[term.toLowerCase()]={term:term,definition:defn,domain:domain||'general'};return db;}
function termLookup(db,term){return db.terms[term.toLowerCase()]||null;}
function termByDomain(db,domain){return Object.keys(db.terms).map(function(k){return db.terms[k];}).filter(function(t){return t.domain===domain;});}
function termSuggest(db,prefix){return Object.keys(db.terms).filter(function(k){return k.indexOf(prefix.toLowerCase())===0;}).map(function(k){return db.terms[k].term;});}

Demo: Domain filter

var db=termDb();termAdd(db,'a','x','sw');termAdd(db,'b','y','hw');var s=termByDomain(db,'sw');J({n:s.length});assert(s.length===1&&s[0].term==='a','domain');
