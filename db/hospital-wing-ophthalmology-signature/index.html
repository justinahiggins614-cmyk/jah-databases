# Signature Clear Vision Bell — Signature design prototype
def snellen_to_logmar(num, den=20):
    """Converts Snellen fraction to logMAR for tracking vision change."""
    import math
    logmar = round(-math.log10(num / den), 2)
    band = "normal" if logmar <= 0.0 else "mild loss" if logmar <= 0.3 else "moderate loss" if logmar <= 1.0 else "severe loss"
    return f"{num}/{den} = {logmar} logMAR ({band})"
print(snellen_to_logmar(20, 200))
